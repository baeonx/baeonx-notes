# stringstream&

```cpp
class Logger {
    std::stringstream m_ss;
public:
    // 返回类型是std::stringstream&
    // 有引号，因为返回的是引用不是拷贝
    std::stringstream& getSS() { return m_ss; }
};

// 使用
logger.getSS() << "Error: " << errCode << " at " << __LINE__;
```







```cpp
class LogStream {
    std::stringstream m_ss;
public:
    LogStream& operator<<(const std::string& msg) {
        m_ss << msg;
        return *this;
    }
    
    std::stringstream& getSS() { return m_ss; }
    
    void flush() {
        std::cout << m_ss.str();
        m_ss.str(""); // 清空
    }
};


LogStream log;
// 现在可以这样使用
log << "Hello" << " " << "World";  // 链式调用

// 相当于
log.operator<<("Hello").operator<<(" ").operator<<("World");
log.flush();  // 输出：Hello World
```
