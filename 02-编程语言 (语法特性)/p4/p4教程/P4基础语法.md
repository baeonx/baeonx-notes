**v1model 完整流水线**

```
┌──────────┐ ┌──────────────┐ ┌──────────┐ ┌─────────┐ ┌───────────────┐ ┌──────────┐
│  parser  │→│ verify       │→│ ingress  │→│ egress  │→│ compute       │→│ deparser │
│          │ │ checksum     │ │          │ │         │ │ checksum      │ │          │
└──────────┘ └──────────────┘ └──────────┘ └─────────┘ └───────────────┘ └──────────┘
     ↑              ↑              ↑           ↑              ↑              ↑
   第1个参数        第2个参数        第3个参数    第4个参数        第5个参数        第6个参数
```

```
包进入 → [Parser] → [Verify] → [Ingress] → [Egress] → [Compute] → [Deparser] → 包发出
```


```cpp
// ============================================================
// 第1部分：引入核心库
// ============================================================
#include <core.p4>
#include <v1model.p4>


// ============================================================
// 第2部分：定义 header 和 struct
// ============================================================

header ethernet_h {
    bit<48> dstAddr;
    bit<48> srcAddr;
    bit<16> etherType;
}

header ipv4_h {
    bit<4> version;
    bit<4> ihl;
    bit<8> diffserv;
    bit<16> totalLen;
    bit<16> identification;
    bit<3> flags;
    bit<13> fragOffset;
    bit<8> ttl;
    bit<8> protocol;
    bit<16> hdrChecksum;
    bit<32> srcAddr;
    bit<32> dstAddr;
}

header tcp_h {
    bit<16> srcPort;
    bit<16> dstPort;
    bit<32> seqNo;
    bit<32> ackNo;
    bit<4> dataOffset;
    bit<3> flags;
    bit<16> window;
    bit<16> checksum;
    bit<16> urgentPtr;
}

header udp_h {
    bit<16> srcPort;
    bit<16> dstPort;
    bit<16> length;
    bit<16> checksum;
}

header arp_h {
    bit<16> hwType;
    bit<16> protoType;
    bit<8> hwAddrLen;
    bit<8> protoAddrLen;
    bit<16> opcode;
    bit<48> senderMac;
    bit<32> senderIp;
    bit<48> targetMac;
    bit<32> targetIp;
}

struct headers {
    ethernet_h ethernet;
    ipv4_h ipv4;
    tcp_h tcp;
    udp_h udp;
    arp_h arp;
}

struct metadata {
    bit<32> drop_count;
    bit<9> ingress_port;
    bit<9> egress_port;
    bit<32> egress_count;
}


// ============================================================
// 第3部分：Parser（解析器）
// ============================================================

parser MyParser(packet_in b, out headers hdr, inout metadata meta) {
    
    state start {
        b.extract(hdr.ethernet);
        transition select(hdr.ethernet.etherType) {
            0x0800: parse_ipv4;
            0x0806: parse_arp;
            default: accept;
        }
    }

    state parse_ipv4 {
        b.extract(hdr.ipv4);
        verify(hdr.ipv4.version == 4w4, error.IPv4IncorrectVersion);
        verify(hdr.ipv4.ihl == 4w5, error.IPv4OptionsNotSupported);
        transition select(hdr.ipv4.protocol) {
            8w6 : parse_tcp;
            8w17: parse_udp;
            _   : accept;
        }
    }

    state parse_tcp {
        b.extract(hdr.tcp);
        transition accept;
    }

    state parse_udp {
        b.extract(hdr.udp);
        transition accept;
    }

    state parse_arp {
        b.extract(hdr.arp);
        transition accept;
    }
}


// ============================================================
// 第4部分：Checksum 校验（VerifyChecksum）
// ============================================================

control MyVerifyChecksum(inout headers hdr, inout metadata meta) {
    Checksum16 ip_checksum = Checksum16();
    apply {
        if (hdr.ipv4.isValid()) {
            ip_checksum.clear();
            ip_checksum.update(hdr.ipv4);
            verify(ip_checksum.get() == 16w0, error.IPv4ChecksumError);
        }
    }
}


// ============================================================
// 第5部分：Ingress Control（入口处理）
// ============================================================

control MyIngress(inout headers hdr,
                  inout metadata meta,
                  inout standard_metadata_t standard_metadata) {
    
    // -------------------- 定义 action --------------------
    
    action forward(bit<9> port) {
        standard_metadata.egress_spec = port;
    }

    action drop() {
        standard_metadata.egress_spec = 0xFFFF;
    }

    action NoAction() { }

    action set_dmac_and_forward(bit<48> mac, bit<9> port) {
        hdr.ethernet.dstAddr = mac;
        standard_metadata.egress_spec = port;
    }

    action set_smac(bit<48> mac) {
        hdr.ethernet.srcAddr = mac;
    }

    action set_dmac(bit<48> mac) {
        hdr.ethernet.dstAddr = mac;
    }

    // -------------------- 定义 table --------------------
    
    table mac_table {
        key = {
            hdr.ethernet.dstAddr : exact;
        }
        actions = {
            forward;
            set_dmac_and_forward;
            set_dmac;
            drop;
            NoAction;
        }
        const entries = {
            (0x112233445566) : forward(1);
            (0xaabbccddee02) : set_dmac_and_forward(0x112233445566, 2);
            (0xffffffffffff) : drop();
            (0x001122334455) : set_dmac(0x5566778899aa);
        }
        default_action = NoAction();
        size = 1024;
    }

    table ipv4_table {
        key = {
            hdr.ipv4.dstAddr : lpm;
        }
        actions = {
            forward;
            drop;
            NoAction;
        }
        const entries = {
            (192.168.1.0/24) : forward(1);
            (10.0.0.0/8) : forward(2);
            (172.16.0.0/12) : forward(3);
            (0.0.0.0/0) : drop();
        }
        default_action = NoAction();
        size = 2048;
    }

    table tcp_table {
        key = {
            hdr.tcp.dstPort : exact;
        }
        actions = {
            forward;
            drop;
            NoAction;
        }
        const entries = {
            (16w80) : forward(3);
            (16w443) : forward(3);
            (16w22) : forward(4);
            (16w23) : forward(4);
            (16w25) : forward(5);
        }
        default_action = NoAction();
        size = 512;
    }

    table udp_table {
        key = {
            hdr.udp.dstPort : exact;
        }
        actions = {
            forward;
            drop;
            NoAction;
        }
        const entries = {
            (16w53) : forward(5);
            (16w123) : forward(6);
            (16w67) : forward(7);
            (16w68) : forward(7);
            (16w161) : forward(8);
        }
        default_action = NoAction();
        size = 512;
    }

    // -------------------- apply 执行 --------------------
    
    apply {
        var mac_result = mac_table.apply();
        
        if (hdr.ipv4.isValid()) {
            ipv4_table.apply();
        }
        
        if (hdr.tcp.isValid()) {
            tcp_table.apply();
        } else if (hdr.udp.isValid()) {
            udp_table.apply();
        }
        
        if (!mac_result.hit) {
            meta.drop_count = meta.drop_count + 1;
        }
        
        meta.ingress_port = standard_metadata.ingress_port;
    }
}


// ============================================================
// 第6部分：Egress Control（出口处理）
// ============================================================

control MyEgress(inout headers hdr,
                 inout metadata meta,
                 inout standard_metadata_t standard_metadata) {
    
    // -------------------- 定义 action --------------------
    
    action forward(bit<9> port) {
        standard_metadata.egress_spec = port;
    }

    action drop() {
        standard_metadata.egress_spec = 0xFFFF;
    }

    action NoAction() { }

    action set_egress_by_ingress(bit<9> port) {
        standard_metadata.egress_spec = port;
    }

    action set_egress_smac(bit<48> mac) {
        hdr.ethernet.srcAddr = mac;
    }

    action set_egress_dmac(bit<48> mac) {
        hdr.ethernet.dstAddr = mac;
    }

    action decrement_ttl() {
        hdr.ipv4.ttl = hdr.ipv4.ttl - 1;
    }

    // -------------------- 定义 table --------------------
    
    table egress_port_table {
        key = {
            meta.ingress_port : exact;
        }
        actions = {
            set_egress_by_ingress;
            NoAction;
        }
        const entries = {
            (9w1) : set_egress_by_ingress(2);
            (9w2) : set_egress_by_ingress(1);
            (9w3) : set_egress_by_ingress(4);
            (9w4) : set_egress_by_ingress(3);
        }
        default_action = NoAction();
        size = 64;
    }

    table egress_mac_rewrite {
        key = {
            hdr.ethernet.dstAddr : exact;
        }
        actions = {
            set_egress_smac;
            set_egress_dmac;
            NoAction;
        }
        const entries = {
            (0x112233445566) : set_egress_smac(0x001122334455);
            (0xaabbccddee02) : set_egress_smac(0x5566778899aa);
            (0xffffffffffff) : set_egress_dmac(0x010203040506);
        }
        default_action = NoAction();
        size = 128;
    }

    table egress_ttl_table {
        key = {
            hdr.ipv4.dstAddr : lpm;
        }
        actions = {
            decrement_ttl;
            NoAction;
        }
        const entries = {
            (192.168.1.0/24) : decrement_ttl();
            (10.0.0.0/8) : decrement_ttl();
            (0.0.0.0/0) : NoAction();
        }
        default_action = NoAction();
        size = 256;
    }

    // -------------------- apply 执行 --------------------
    
    apply {
        egress_port_table.apply();
        egress_mac_rewrite.apply();
        
        if (hdr.ipv4.isValid()) {
            egress_ttl_table.apply();
        }
        
        meta.egress_count = meta.egress_count + 1;
        meta.egress_port = standard_metadata.egress_spec;
        
        if (meta.ingress_port == 9w1 && standard_metadata.egress_spec == 2) {
            drop();
        }
    }
}


// ============================================================
// 第7部分：Checksum 计算（ComputeChecksum）
// ============================================================

control MyComputeChecksum(inout headers hdr, inout metadata meta) {
    Checksum16 ip_checksum = Checksum16();
    apply {
        if (hdr.ipv4.isValid()) {
            hdr.ipv4.hdrChecksum = 16w0;
            ip_checksum.clear();
            ip_checksum.update(hdr.ipv4);
            hdr.ipv4.hdrChecksum = ip_checksum.get();
        }
    }
}


// ============================================================
// 第8部分：Deparser（发报器）
// ============================================================

control MyDeparser(packet_out b, in headers hdr) {
    apply {
        b.emit(hdr.ethernet);
        b.emit(hdr.ipv4);
        b.emit(hdr.tcp);
        b.emit(hdr.udp);
        b.emit(hdr.arp);
    }
}


// ============================================================
// 第9部分：Package 实例化（程序入口）
// ============================================================

V1Switch(
    MyParser(),
    MyVerifyChecksum(),
    MyIngress(),
    MyEgress(),
    MyComputeChecksum(),
    MyDeparser()
) main;
```

### 第2部分：定义 header 和 struct

**以太网头（14字节）**

```cpp
header ethernet_h {
    bit<48> dstAddr;      // 目的MAC地址
    bit<48> srcAddr;      // 源MAC地址
    bit<16> etherType;    // 以太网类型（0x0800=IPv4, 0x0806=ARP）
}
```



**IPv4头（20字节）**

```cpp
header ipv4_h {
    bit<4> version;        // IP版本号（必须是4）
    bit<4> ihl;            // 头部长度（单位：4字节，5表示20字节）
    bit<8> diffserv;       // 区分服务
    bit<16> totalLen;      // 总长度
    bit<16> identification; // 标识符
    bit<3> flags;          // 标志位
    bit<13> fragOffset;    // 分片偏移
    bit<8> ttl;            // 生存时间（每跳减1）
    bit<8> protocol;       // 上层协议（6=TCP, 17=UDP）
    bit<16> hdrChecksum;   // 头部校验和
    bit<32> srcAddr;       // 源IP地址
    bit<32> dstAddr;       // 目的IP地址
}
```



**TCP头（20字节）**

```cpp
header tcp_h {
    bit<16> srcPort;       // 源端口
    bit<16> dstPort;       // 目的端口
    bit<32> seqNo;         // 序列号
    bit<32> ackNo;         // 确认号
    bit<4> dataOffset;     // 数据偏移
    bit<3> flags;          // 标志位（SYN, ACK, FIN等）
    bit<16> window;        // 窗口大小
    bit<16> checksum;      // 校验和
    bit<16> urgentPtr;     // 紧急指针
}
```



**UDP头（8字节）**

```cpp
header udp_h {
    bit<16> srcPort;       // 源端口
    bit<16> dstPort;       // 目的端口
    bit<16> length;        // UDP长度
    bit<16> checksum;      // 校验和
}
```



**ARP头（28字节）**

```cpp
header arp_h {
    bit<16> hwType;        // 硬件类型（以太网=1）
    bit<16> protoType;     // 协议类型（IPv4=0x0800）
    bit<8> hwAddrLen;      // 硬件地址长度（MAC=6）
    bit<8> protoAddrLen;   // 协议地址长度（IP=4）
    bit<16> opcode;        // 操作码（1=请求, 2=应答）
    bit<48> senderMac;     // 发送者MAC
    bit<32> senderIp;      // 发送者IP
    bit<48> targetMac;     // 目标MAC
    bit<32> targetIp;      // 目标IP
}
```

### **第3部分：Parser（解析器）**

**解析流程图**

```text
数据包进入
    ↓
┌─────────────────────────────────────────────┐
│ state start                                 │
│ extract(ethernet) ← 从报文读14字节            │
│ select(etherType)                           │
│   ├── 0x0800 → parse_ipv4                   │
│   ├── 0x0806 → parse_arp                    │
│   └── default → accept（不继续解析）           │
└─────────────────────────────────────────────┘
                    ↓
┌─────────────────────────────────────────────┐
│ state parse_ipv4                            │
│ extract(ipv4) ← 读20字节                      │
│ verify(version == 4) → 否则丢弃               │
│ verify(ihl == 5) → 否则丢弃                   │
│ select(protocol)                            │
│   ├── 6 → parse_tcp                         │
│   ├── 17 → parse_udp                        │
│   └── _ → accept                            │
└─────────────────────────────────────────────┘
                    ↓
┌─────────────────────────────────────────────┐
│ state parse_tcp / parse_udp / parse_arp    │
│ extract(对应头部)                          │
│ accept                                     │
└─────────────────────────────────────────────┘
                    ↓
              accept（解析完成）
```

**关键语法**

| 语法                                    | 作用                               |
| :-------------------------------------- | :--------------------------------- |
| `b.extract(hdr.xxx)`                    | 从原始报文读取数据，填入header结构 |
| `verify(条件, 错误码)`                  | 条件为false时丢弃报文              |
| `transition select(值) { 匹配: 状态; }` | 根据值跳转到不同状态               |
| `default`                               | 匹配所有其他值                     |
| `_`                                     | 通配符，匹配任意值                 |
| `accept`                                | 系统预定义状态，解析成功           |
| `reject`                                | 系统预定义状态，解析失败           |

### **第4部分：VerifyChecksum（校验和验证）**

```cpp
control MyVerifyChecksum(inout headers hdr, inout metadata meta) {
    Checksum16 ip_checksum = Checksum16();
    apply {
        if (hdr.ipv4.isValid()) {
            ip_checksum.clear();
            ip_checksum.update(hdr.ipv4);
            verify(ip_checksum.get() == 16w0, error.IPv4ChecksumError);
        }
    }
}
```

**执行逻辑**

```text
1. 检查 hdr.ipv4 是否有效（valid == true）
   ↓
2. ip_checksum.clear()  ← 清空校验和计算器
   ↓
3. ip_checksum.update(hdr.ipv4)  ← 将整个IPv4头喂入计算器
   ↓
4. ip_checksum.get()  ← 获取计算结果
   ↓
5. verify(结果 == 16w0)  ← 正确的校验和应为0
   ↓
   ├── 成立 → 继续
   └── 不成立 → 丢弃报文
```



**校验原理：** 把 `hdrChecksum` 字段一起纳入计算，正确结果为 `0`。

### 第5部分：Ingress Control（入口处理）

**Actions（动作）**

| Action                            | 参数          | 作用               |
| :-------------------------------- | :------------ | :----------------- |
| `forward(port)`                   | `bit<9> port` | 设置出端口         |
| `drop()`                          | 无            | 设置丢弃（0xFFFF） |
| `NoAction()`                      | 无            | 什么都不做         |
| `set_dmac_and_forward(mac, port)` | MAC地址, 端口 | 改目的MAC + 转发   |
| `set_smac(mac)`                   | MAC地址       | 改源MAC            |
| `set_dmac(mac)`                   | MAC地址       | 改目的MAC          |

**Tables（匹配-动作表）**

**1. mac_table（MAC表）**

```cpp
key = { hdr.ethernet.dstAddr : exact; }    // 精确匹配目的MAC
actions = { forward; set_dmac_and_forward; set_dmac; drop; NoAction; }
```

| 匹配值           | 动作                                      | 效果                               |
| :--------------- | :---------------------------------------- | :--------------------------------- |
| `0x112233445566` | `forward(1)`                              | 从端口1转发                        |
| `0xaabbccddee02` | `set_dmac_and_forward(0x112233445566, 2)` | 改MAC为0x112233445566，从端口2转发 |
| `0xffffffffffff` | `drop()`                                  | 丢弃                               |
| `0x001122334455` | `set_dmac(0x5566778899aa)`                | 改目的MAC为0x5566778899aa          |



**2. ipv4_table（IPv4路由表）**

```
key = { hdr.ipv4.dstAddr : lpm; }          // 最长前缀匹配
actions = { forward; drop; NoAction; }
```

| 匹配值           | 动作         | 效果                    |
| :--------------- | :----------- | :---------------------- |
| `192.168.1.0/24` | `forward(1)` | 发往192.168.1.x → 端口1 |
| `10.0.0.0/8`     | `forward(2)` | 发往10.x.x.x → 端口2    |
| `172.16.0.0/12`  | `forward(3)` | 发往172.16.x.x → 端口3  |
| `0.0.0.0/0`      | `drop()`     | 默认路由 → 丢弃         |

**LPM（最长前缀匹配）：** 如果有 `192.168.1.0/24` 和 `192.168.1.5/32` 同时匹配，选择更精确的（/32）。

**3. tcp_table（TCP端口表）**

```cpp
key = { hdr.tcp.dstPort : exact; }
```

| 目的端口    | 动作         | 效果  |
| :---------- | :----------- | :---- |
| 80 (HTTP)   | `forward(3)` | 端口3 |
| 443 (HTTPS) | `forward(3)` | 端口3 |
| 22 (SSH)    | `forward(4)` | 端口4 |
| 23 (Telnet) | `forward(4)` | 端口4 |
| 25 (SMTP)   | `forward(5)` | 端口5 |

**4. udp_table（UDP端口表）**

```cpp
key = { hdr.udp.dstPort : exact; }
```

| 目的端口     | 动作         | 效果  |
| :----------- | :----------- | :---- |
| 53 (DNS)     | `forward(5)` | 端口5 |
| 123 (NTP)    | `forward(6)` | 端口6 |
| 67/68 (DHCP) | `forward(7)` | 端口7 |
| 161 (SNMP)   | `forward(8)` | 端口8 |



**Ingress apply 执行流程**

```cpp
apply {
    // 第1步：查MAC表（保存结果用于统计）
    var mac_result = mac_table.apply();
    
    // 第2步：查IPv4表
    if (hdr.ipv4.isValid()) {
        ipv4_table.apply();
    }
    
    // 第3步：查TCP/UDP表
    if (hdr.tcp.isValid()) {
        tcp_table.apply();
    } else if (hdr.udp.isValid()) {
        udp_table.apply();
    }
    
    // 第4步：统计MAC未命中
    if (!mac_result.hit) {
        meta.drop_count = meta.drop_count + 1;
    }
    
    // 第5步：记录入端口（供Egress使用）
    meta.ingress_port = standard_metadata.ingress_port;
}
```



**执行顺序图：**

```text
数据包
    ↓
mac_table.apply() → 根据目的MAC设置出端口
    ↓
ipv4_table.apply() → 根据目的IP设置出端口（覆盖）
    ↓
tcp_table.apply() / udp_table.apply() → 根据端口设置出端口（再覆盖）
    ↓
最终出端口 = 最后修改的值
```

### 第6部分：Egress Control（出口处理）

**Actions（动作）**

| Action                        | 参数 | 作用                 |
| :---------------------------- | :--- | :------------------- |
| `forward(port)`               | 端口 | 设置出端口           |
| `drop()`                      | 无   | 丢弃                 |
| `NoAction()`                  | 无   | 什么都不做           |
| `set_egress_by_ingress(port)` | 端口 | 根据入端口设置出端口 |
| `set_egress_smac(mac)`        | MAC  | 出口方向修改源MAC    |
| `set_egress_dmac(mac)`        | MAC  | 出口方向修改目的MAC  |
| `decrement_ttl()`             | 无   | TTL减1               |



**Tables**

**1. egress_port_table（端口映射表）**

```cpp
key = { meta.ingress_port : exact; }
```

| 入端口 | 动作                       | 效果 |
| :----- | :------------------------- | :--- |
| 1      | `set_egress_by_ingress(2)` | 1→2  |
| 2      | `set_egress_by_ingress(1)` | 2→1  |
| 3      | `set_egress_by_ingress(4)` | 3→4  |
| 4      | `set_egress_by_ingress(3)` | 4→3  |

**2. egress_mac_rewrite（MAC重写表）**

```cpp
key = { hdr.ethernet.dstAddr : exact; }
```

| 目的MAC          | 动作                              | 效果                      |
| :--------------- | :-------------------------------- | :------------------------ |
| `0x112233445566` | `set_egress_smac(0x001122334455)` | 源MAC改为0x001122334455   |
| `0xaabbccddee02` | `set_egress_smac(0x5566778899aa)` | 源MAC改为0x5566778899aa   |
| `0xffffffffffff` | `set_egress_dmac(0x010203040506)` | 目的MAC改为0x010203040506 |

**3. egress_ttl_table（TTL表）**

```cpp
key = { hdr.ipv4.dstAddr : lpm; }
```

| 目的网段         | 动作              |
| :--------------- | :---------------- |
| `192.168.1.0/24` | `decrement_ttl()` |
| `10.0.0.0/8`     | `decrement_ttl()` |
| `0.0.0.0/0`      | `NoAction()`      |

**Egress apply 执行流程**

```cpp
apply {
    // 第1步：端口映射
    egress_port_table.apply();

    // 第2步：MAC重写
    egress_mac_rewrite.apply();
    
    // 第3步：TTL处理
    if (hdr.ipv4.isValid()) {
        egress_ttl_table.apply();
    }
    
    // 第4步：统计
    meta.egress_count = meta.egress_count + 1;
    meta.egress_port = standard_metadata.egress_spec;
    
    // 第5步：出口过滤
    if (meta.ingress_port == 9w1 && standard_metadata.egress_spec == 2) {
        drop();
    }
}
```

### 第7部分：ComputeChecksum（校验和计算）

```cpp
control MyComputeChecksum(inout headers hdr, inout metadata meta) {
    Checksum16 ip_checksum = Checksum16();
    apply {
        if (hdr.ipv4.isValid()) {
            // 1. 先清零
            hdr.ipv4.hdrChecksum = 16w0;
            // 2. 重新计算
            ip_checksum.clear();
            ip_checksum.update(hdr.ipv4);
            // 3. 写回结果
            hdr.ipv4.hdrChecksum = ip_checksum.get();
        }
    }
}
```

**为什么需要重新计算？**

Ingress/Egress 可能修改了IPv4头（如TTL、MAC地址等），修改后校验和必须重新计算，否则接收端会认为校验和错误而丢弃报文。

### 第8部分：Deparser（发报器）

```cpp
control MyDeparser(packet_out b, in headers hdr) {
    apply {
        b.emit(hdr.ethernet);
        b.emit(hdr.ipv4);
        b.emit(hdr.tcp);
        b.emit(hdr.udp);
        b.emit(hdr.arp);
    }
}
```



**emit 行为**

| 头部           | 是否添加                     |
| :------------- | :--------------------------- |
| `hdr.ethernet` | 总是添加（以太网头始终存在） |
| `hdr.ipv4`     | 仅当 `valid == true` 时添加  |
| `hdr.tcp`      | 仅当 `valid == true` 时添加  |
| `hdr.udp`      | 仅当 `valid == true` 时添加  |
| `hdr.arp`      | 仅当 `valid == true` 时添加  |

**组装顺序就是 emit 顺序，决定报文中字段的排列顺序。**

## 第3章 P4语言详解 — 完整详细总结

## 3.1 P4语言概述

### 3.1.1 P4语言的关键字

P4语言定义了40个关键字，如表3-1所示。

**表3-1 P4关键字**

| 分类           | 关键字                                                       |
| :------------- | :----------------------------------------------------------- |
| **数据类型**   | `action`、`bit`、`bool`、`control`、`enum`、`error`、`false`、`header`、`header_union`、`int`、`match_kind`、`package`、`parser`、`state`、`struct`、`table`、`true`、`tuple`、`value_set`、`varbit`、`void` |
| **类型修饰符** | `abstract`、`const`、`extern`、`in`、`inout`、`out`、`type`、`typedef` |
| **语句**       | `else`、`exit`、`if`、`return`、`select`、`switch`、`transition` |
| **操作**       | `apply`、`verify`                                            |
| **其他**       | `default`、`this`                                            |

**与C语言相似的关键字：** `if`、`else`、`return`、`bool`、`false`、`true`、`void`、`int`、`const`、`struct`

**P4新增的关键字：** `header`、`table`、`action`、`apply`、`control`、`parser`（针对报文处理）

### 3.1.2 P4语言的数据类型

P4是**强类型语言**，所有数据都需要定义类型，表达式的求值规则与C语言类似。

## 3.2 P4语言基本数据类型及其表达式

P4语言的基本数据类型主要包括**整型**和**布尔类型**。P4不包含字符类型、字符串类型。

### 3.2.1 无符号整型（`bit<W>`）

无符号整型是P4编程中最常用的数据类型。例如IPv4的源地址/目的地址是32位无符号整型，TCP的源端口/目的端口是16位无符号整型。

**1. 无符号整型的定义**

```cpp
bit<W> v;
```

| 部分  | 含义                       |
| :---- | :------------------------- |
| `bit` | 关键字，表示无符号整型     |
| `W`   | 位宽（常量表达式，正整数） |
| `v`   | 变量名                     |

**位串概念：** `bit<W>` 表示一个长度为W的位序列，从0到W-1编号，第0位是最低位，第W-1位是最高有效位。

**示例：**

```cpp
bit<8> v = 0x80;  // 第0位是0，第7位是1
```



**注意：** P4中没有`unsigned`关键字，用`bit<W>`表示无符号。

**位宽限制：** 虽然语法支持任意位宽，但实际受Target限制。例如p4c/BMv2最大支持2048位。

```cpp
bit<800000> v = 0x80;  // ❌ 报错：Compiler only supports widths up to 2048
```



**2. 无符号整型支持的操作**

**（1）赋值操作**

| 规则     | 说明                 |
| :------- | :------------------- |
| 位宽相同 | 可以直接赋值         |
| 位宽不同 | 必须显式强制类型转换 |

**（2）算术运算**

| 操作     | 运算符 | 说明                                                      |
| :------- | :----- | :-------------------------------------------------------- |
| 加法     | `+`    | 结果位宽与操作数相同，超出的高位被截断                    |
| 减法     | `-`    | 第二个操作数取反后与第一个相加                            |
| 乘法     | `*`    | 结果位宽与操作数相同，高位截断；某些Target只允许2的幂次方 |
| 饱和加法 | `|+`   | 结果超过位宽最大值则取最大值（如8位结果真实值258→255）    |
| 饱和减法 | `-|`   | 结果低于位宽最小值则取最小值                              |
| 除法     | 未定义 | P4规范未定义，需参考具体编译器手册                        |

**示例：**

```cpp
bit<8> a = 200;
bit<8> b = 100;
bit<8> c = a + b;  // 真实值300，但8位最大255，所以c = 44（300-256）
```



**（3）位运算**

| 操作     | 运算符 | 要求                   |
| :------- | :----- | :--------------------- |
| 按位与   | `&`    | 相同宽度               |
| 按位或   | `|`    | 相同宽度               |
| 按位取反 | `~`    | 单操作数               |
| 按位异或 | `^`    | 相同宽度               |
| 逻辑左移 | `<<`   | 右操作数非负整型字面量 |
| 逻辑右移 | `>>`   | 右操作数非负整型字面量 |

**移位规则：** 如果移动位数超过左操作数的宽度，计算结果为全0。

**（4）关系运算**

| 操作     | 运算符 | 要求     | 结果类型 |
| :------- | :----- | :------- | :------- |
| 等于     | `==`   | 相同宽度 | `bool`   |
| 不等于   | `!=`   | 相同宽度 | `bool`   |
| 大于     | `>`    | 相同宽度 | `bool`   |
| 小于     | `<`    | 相同宽度 | `bool`   |
| 大于等于 | `>=`   | 相同宽度 | `bool`   |
| 小于等于 | `<=`   | 相同宽度 | `bool`   |

**重要规则：** 宽度不同不允许比较，除非显式类型转换。

```cpp
bit<8> a = 5;
bit<16> b = 5;
if (a == b) { }  // ❌ 错误！宽度不同
if (a == (bit<8>)b) { }  // ✅ 正确
```



**（5）拼接运算（`++`）**

把两个位串连接起来，结果长度 = 两操作数长度之和，最高有效位取自左操作数。

```cpp
bit<4> a = (bit<4>)(0xF);    // a = 1111
bit<8> b = (bit<8>)(0xF0);   // b = 11110000
bit<12> c = a ++ b;           // c = 111111110000 = 0xFF0
```



**（6）切片运算（`[n:m]`）**

从一个位串中提取一段连续的子串，`n>=m`，结果长度为 `n-m+1`，**包含两端端点**。

```cpp
// 设置TCP SYN标志位（flags[1]表示SYN位）
void tcp_set_syn_flag(inout headers hdr) {
    apply {
        hdr.tcp.flags[1:1] = 1;
    }
}
```



**切片可以作为左值：**

```cpp
e[n:m] = x;   // 将x赋值给e的n到m位
```



### 3.2.2 有符号整型

有符号整型分为两类：**固定宽度有符号整型**和**无限精度整型**。

**1. 固定宽度有符号整型的定义**

```cpp
int<W> v;
```

| 部分     | 含义                   |
| :------- | :--------------------- |
| `int<W>` | 位宽为W的有符号整型    |
| W        | 常量表达式，正整数     |
| 位编号   | 0~W-1，第W-1位是符号位 |

**示例：**

```cpp
int<64> v;  // 64位有符号，第63位是符号位
```



**2. 固定宽度有符号整型支持的操作**

基本与无符号整型类似，但移位有特殊规则：

**算术移位规则：**

| 移位方向 | 规则                                     |
| :------- | :--------------------------------------- |
| 左移     | 低位补0，符号位不变                      |
| 右移     | 高位补**符号位**（算术移位），符号位不变 |

**移位超出宽度的情况：**

| 情况         | 结果       |
| :----------- | :--------- |
| 左移超过宽度 | 所有bit为0 |
| 右移超过宽度 | 所有bit为0 |
| 右移负数     | 所有bit为1 |

**拼接运算：**

```cpp
int<4> a = (int<4>)(0x7);    // a = 0111
int<8> b = (int<8>)(0x70);   // b = 01110000
int<12> c = a ++ b;           // c = 011101110000 = 0x770
```



**3. 无限精度整型的定义**

```cpp
int v;
```

| 项目     | 说明                               |
| :------- | :--------------------------------- |
| 位宽     | **无限长**（可以表示无限大的数值） |
| 使用场景 | **仅编译时常量**                   |
| 运行时   | **禁止**作为运行时变量             |

**正确用法（编译时常量）：**

```cpp
const int a = 5;
const int b = 2 * a;
const int c = b - a + 3;
```

**错误用法（运行时变量）：**

```cpp
int a = 5;  // ❌ 编译错误！Cannot declare variables with type int
```

报错信息：

```cpp
error: a: Cannot declare variables with type int
    int a = 5;
    ^
```



**4. 无限精度整型支持的操作**

与固定宽度有符号整型类似，但有以下**重要限制**：

| 限制                     | 说明                                    |
| :----------------------- | :-------------------------------------- |
| 不能与 `int<W>` 混合运算 | 除非强制类型转换                        |
| 不支持位运算             | `&`、`|`、`~`、`^`、`<<`、`>>` 都不支持 |
| 不支持饱和运算           | `|+`、`-|` 不支持                       |

### 3.2.3 整型常量（整型字面量）

整型常量有三种类型：

| 类型                  | 格式       | 示例             | 说明           |
| :-------------------- | :--------- | :--------------- | :------------- |
| 类型1：简单整型常量   | 直接写数字 | `10`             | 类型为`int`    |
| 类型2：无符号整型常量 | `Nw值`     | `8w10`、`32w0x0` | 类型为`bit<N>` |
| 类型3：有符号整型常量 | `Ns值`     | `8s10`           | 类型为`int<N>` |

**示例表：**

| 整型常量 | 含义                                   | 分类  |
| :------- | :------------------------------------- | :---- |
| `10`     | 类型是`int`，值是10                    | 类型1 |
| `8w10`   | 类型是`bit<8>`，值是10                 | 类型2 |
| `1w10`   | 类型是`bit<1>`，值是**0**（溢出警告）  | 类型2 |
| `8s10`   | 类型是`int<8>`，值是10                 | 类型3 |
| `2s3`    | 类型是`int<2>`，值是**-1**（溢出警告） | 类型3 |
| `1s1`    | 类型是`int<1>`，值是**-1**（溢出警告） | 类型3 |

**使用示例：**

```cpp
#define ZERO (32w0x0)    // 宏定义
bit<32> b = 32w10;       // 变量初始化
```



### 3.2.4 varbit 类型（可变长度位串）

某些网络协议字段长度是可变的（如IPv4选项字段），P4提供`varbit`类型支持。

**1. varbit类型的定义**

```cpp
varbit<W> val;
```

| 术语     | 含义              |
| :------- | :---------------- |
| 静态位宽 | `W`，最大可能长度 |
| 动态位宽 | 运行时实际长度    |

**示例：**

```cpp
varbit<40> tcp_option;  // 位宽可能为0~40位
```

**Target限制：**

- 可能限制位宽范围
- 可能要求动态位宽必须是字节（8bit）对齐
- 有些Target可能完全不支持`varbit`



**2. varbit类型支持的操作**

| 操作 | 规则                                                         |
| :--- | :----------------------------------------------------------- |
| 赋值 | 两个varbit变量的**静态位宽必须相同**；赋值时同时复制数据和动态位宽 |
| 比较 | 两个varbit变量的**静态位宽相同**时，动态位宽相等且所有位值相等才相等 |

**使用场景：**

1. **Parser中的`extract()`**：把数据提取到包含varbit字段的头部，自动设置动态位宽
2. **Deparser中的`emit()`**：根据动态位宽提取有效数据并发送

### 3.2.5 布尔类型（`bool`）

**1. 布尔类型的定义**

```cpp
bool is_ipv4_packet = true;
```

布尔值不是整型，但可以通过**强制类型转换**转换为整型。

**2. 布尔类型支持的操作**

| 操作     | 运算符 | 说明         |
| :------- | :----- | :----------- |
| 逻辑与   | `&&`   | 支持短路求值 |
| 逻辑或   | `||`   | 支持短路求值 |
| 逻辑非   | `!`    | 单操作数     |
| 相等比较 | `==`   | 结果bool     |
| 不等比较 | `!=`   | 结果bool     |

**与C语言的关键区别：**

P4**不支持**布尔类型和整型之间的隐式类型转换。

```cpp
// C语言合法
int x;
if (x) { /* 代码省略 */ }

// P4必须写成
int x;
if (x != 0) { /* 代码省略 */ }   // ✅ 正确
if (x) { /* 代码省略 */ }        // ❌ 错误！
```

**条件运算符：** `e1 ? e2 : e3`

- `e1` 必须是布尔类型
- `e2` 和 `e3` 类型必须相同
- 求值规则：先对e1求值，true则对e2求值，否则对e3求值

### 3.2.6 error 类型

**1.error类型的定义**

error类型与C语言的枚举（enum）类似，包含用于指示错误的数值。

```cpp
// P4核心库中的定义
error { ParseError, PacketTooShort }
```

**重要特性：** 多个error声明被编译器合并到统一的error命名空间，与定义位置无关。

**2. error类型支持的操作**

| 操作   | 运算符 |
| :----- | :----- |
| 等于   | `==`   |
| 不等于 | `!=`   |

**示例：**

```cpp
error errorFromParser;
if (errorFromParser != error.NoError) {
    // 发生了错误，处理
}
```

| :----------------- | :---------- | :------------------ | :------ | :--------------------- |

### 3.2.9 与C语言关键区别总结

| 区别点         | C语言          | P4语言                    | 影响                    |
| :------------- | :------------- | :------------------------ | :---------------------- |
| 无符号整型     | `unsigned int` | `bit<W>`                  | 必须明确指定位宽        |
| 位宽           | 依赖平台       | 必须明确指定              | 跨平台一致              |
| `int`类型      | 运行时变量     | **仅编译时常量**          | 运行时不能用`int`       |
| 布尔转整型     | 隐式转换       | **不支持**                | `if(x)`必须写`if(x!=0)` |
| 字符/字符串    | 支持           | **不支持**                | 不需要I/O操作           |
| 指针           | 支持           | **不支持**                | 无内存地址操作          |
| 除法           | 支持           | **规范未定义**            | 需查阅Target手册        |
| 位宽限制       | 无             | 有限制（如p4c最多2048位） | 超大位宽会报错          |
| 移位（有符号） | 实现定义       | 算术移位                  | 结果可预测              |

### 3.2.10 编程注意事项

| 序号 | 注意事项                                               |
| :--- | :----------------------------------------------------- |
| 1    | `bit<W>`和`int<W>`是两个不同类型，**不能直接混合运算** |
| 2    | 宽度不同的位串**不能直接比较**，必须显式转换           |
| 3    | `int`类型**只能用于编译时常量**，不能声明运行时变量    |
| 4    | `bool`不能自动转整型，`if`中必须显式写比较表达式       |
| 5    | 饱和加/减法`|+`和`-|`仅无符号整型支持                  |
| 6    | 切片的两个端点**都包含**在内：`[n:m]`长度为`n-m+1`     |
| 7    | `varbit`类型**静态位宽必须相同**才能赋值/比较          |
| 8    | 整型常量默认是`int`类型，赋值给固定位宽时注意溢出      |
| 9    | 某些Target可能**不支持`varbit`**，使用前需确认         |
| 10   | 位运算要求操作数**宽度相同**，否则需强转               |







## 3.3 复合数据类型及其表达式

### 3.3.1 `enum`: 枚举类型

**1. 枚举类型的定义**

- 使用 `enum` 关键字定义，可包含多个枚举成员，用于提高代码可读性。
- 可以指定成员的值，也可以不指定（编译器自动分配）。
- P4未规定枚举的底层实现和位宽，与编译器和目标平台相关。

**示例（无底层类型）：**

```cpp
enum l3_packet_type_t {
    IPV4 = 0,
    IPV6
}
// 可以指定成员的值，也可以不指定（编译器自动分配）
// IPV4指定了成员的值，IPV6未指定
```

---

**指定底层类型（bit<W> 或 int<W>）：**

```cpp
enum bit<16> ether_type_t {
    VLAN = 0x8100,
    QINQ = 0x9100,
    MPLS = 0x8847,
    IPV4 = 0x0800,
    IPV6 = 0x86dd
}
```

---

**多个符号可映射到同一值（如 `b2` 和 `b3` 都为1）。**

```cpp
enum bit<8> non_unique_t {
    b1 = 0,
    b2 = 1,
    b3 = 1,
    b4 = 2
}
```

**如果枚举项的值超出底层类型的表示范围，编译器将提示警告：**

```cpp
enum bit<8> failing_example_t {
    first = 1,
    second = 2,
    third = 3,
    unrepresentable = 300
}
// 300 无法使用 bit<8> 表示，所以编译器将提示警告。
```

------

**2. 枚举类型支持的操作**

- **无底层类型**：仅支持 `==` 和 `!=` 比较，比较的结果是布尔类型
- **有底层类型**：
  - 可**隐式转换**为底层类型（安全）。
  - 从底层类型转换回枚举必须**显式强制转换**（可能产生未定义值）。

```cpp
// 定义了一个枚举类型 E，其底层类型为 bit<8>
enum bit<8> E {
    e1 = 0,
    e2 = 1,
    e3 = 2
}

bit<8> x;
E a = E.e1;    // ✅ 正确
E a = E.e2;    // ✅ 正确
E a = E.e3;    // ✅ 正确
E a = 3;       // ❌ 错误：不能直接将整数 3 赋值给枚举类型

// 支持 enum bit<8> E==>bit<8>==>enum bit<8> E
E b;
x = (bit<8>)a;    // 隐式转换,将 x 设置为 1; enum bit<8> E==>bit<8>
b = (E)x;    // 将 b 设置为 E.e2; bit<8>==>enum bit<8> E
```

---

因为将枚举类型转换成其底层类型的操作是安全的，所以 P4 语言支持枚举类型到底层类型的隐式类型转换，不需要进行显式地强制类型转换，示例代码如下：

```cpp
// 将 x 设置为 1，E.e2 自动转换为 bit<8>
bit<8> x = E.e2;
E a = E.e2

// 将 y 设置为 8，a 自动转换为 bit<8>，然后进行移位操作
bit<8> y = a << 3;
```

---

但是反过来，从底层类型到枚举类型，则必须使用强制类型转换，不支持隐式类型转换，示例代码如下：

```cpp
enum bit<8> E1 {
    e1 = 0,
    e2 = 1,
    e3 = 2
}

enum bit<8> E2 {
    e1 = 10,
    e2 = 11,
    e3 = 12
}

E1 a = E1.e1;
E2 b = E2.e2;

// 错误: b 可以自动转换为 bit<8>，但是 bit<8> 不能自动转换为 E1
// 因为 a 和 b 是两种不同的枚举类型，它们之间不能直接赋值。
// 类比理解：
// E1 是一种"身份卡"，只能识别 E1 类型的成员（e1、e2、e3）
// E2 是另一种"身份卡"，只能识别 E2 类型的成员（e1、e2、e3）
// 虽然它们都叫 e1、e2、e3，但它们是不同的类型，就像：
// 苹果公司的"苹果" 🍎
// 水果店的"苹果" 🍎
// 虽然都叫"苹果"，但一个是公司品牌，一个是水果，不能混为一谈。
a = b;  // 隐式转换
		// ❌ 类型不匹配！
        // 左边期望 E1 类型，右边是 E2 类型


// b 先自动转换为 bit<8>，值为 11
// 然后通过 (E1) 强制转换为 E1 类型
// 赋值给 a
a = (E1)b; // 显示转换,正确

// 错误: E1.e1 可以自动转换为 bit<8>
// 但是右操作数是 bit<8>，不能自动转换为 E1 类型
a = E1.e1 + 1;

// 正确
// E1.e1 先隐式转换为 bit<8> 类型
// E1.e1 + 1为bit<8>类型
// 将 bit<8> 类型的 1 强制转换为 E1 类型
a = (E1)(E1.e1 + 1);

// 错误
// E1.e1（隐式转换后）	bit<8>0
// E1.e2（隐式转换后）	bit<8>1
// E1.e1 + E1.e2	bit<8> 0 + 1 = 1
// bit<8>无法隐士转换成E1
a = E1.e1 + E1.e2;

// 正确
a = (E1)(E2.e1 + E2.e2);
```

---

注意: 虽然 enum 和底层类型之间的相互转换是安全的，但是把底层类型转换到 enum 时，有可能转换成一个未定义的值。

例如下面这段代码:

```cpp
enum bit<8> E {
    e1 = 0,
    e2 = 1,
    e3 = 2
};

bit<8> x = 5;
E e = (E) x;
```

---

在这段代码里，因为枚举 E 中不存在映射到 5 的符号，所以这个表达式会给 e 设置一个未定义的值。这种转换会引发很多不明显的错误。例如，在下面的代码里，虽然 if...else 的判断条件涵盖了所有枚举成员，但程序还是有可能进入最后的 else 分支，这个分支需要妥善处理。示例代码如下:

```cpp
enum bit<2> MyPartialEnum_t {
    VALUE_A = 2w0,
    VALUE_B = 2w1,
    VALUE_C = 2w2
};

bit<2> y = <some value>;

MyPartialEnum_t x = (MyPartialEnum_t) y;
if (x == MyPartialEnum_t.VALUE_A) {
    // 代码省略
} else if (x == MyPartialEnum_t.VALUE_B) {
    // 代码省略
} else if (x == MyPartialEnum_t.VALUE_C) {
    // 代码省略
} else {
    // 未被定义的值
    // 这个分支的存在是必要的
    // 代码省略
}
```

---

另外，如果 header 里包含枚举类型的字段，在进行 parser 解析的时候一般会使用 transition select 语句匹配所有可能的枚举值，在这种情况下，也需要特别注意 default 分支。当匹配失败时，会进入 default 分支处理，例如下列代码中对 EtherType 的判断：（学完回头再看）

```cpp
enum bit<16> EtherType {
    VLAN = 0x8100,
    IPV4 = 0x0800,
    IPV6 = 0x86dd
}

header ethernet {
    // 其他成员省略
    EtherType etherType;
}

parser my_parser(/* parameters omitted */) {
    state parse_ethernet {
    packet.extract(hdr.ethernet);
    transition select(hdr.ethernet.etherType) {
    EtherType.VLAN : parse_vlan;
    EtherType.IPV4 : parse_ipv4;
    EtherType.IPV6 : parse_ipv6;
    default: reject;
    }
    }
}
```

### 3.3.2 `struct`：结构类型

**1. 结构类型的定义**

- 使用 `struct` 关键字，将多个成员组合成一个类型。
- 允许空结构（无成员）。

**示例：**

```python
struct header_t {
    ethernet_h ethernet;
    ipv4_h ipv4;
    ipv6_h ipv6;
}
```

结构体常用于组织报文头部字段，便于在解析器（parser）和控制逻辑中统一管理。

**2. 结构类型支持的操作**

**1. 结构体赋值**

- **相同类型的结构体**可以直接赋值（整体复制）
- **相同类型且成员可递归比较**，支持 `==` 和 `!=`

**必须同时给所有成员赋值！（与 C 语言不同）**

```cpp
struct S {
    bit<32> a;
    bit<32> b;
}

// 方式1：列表表达式（带字段名）
S s1 = { a = 10, b = 20 };

// 方式2：列表表达式（按顺序）
S s2 = { 10, 20 };

// 方式3：structure-valued 表达式
S s3 = (S) { a = 10, b = 20 };
```

**2. 结构体比较**

**示例 1：相同类型的结构体比较**

```cpp
struct S {
    bit<32> a;
    bit<32> b;
}

S s1 = { a = 10, b = 20 };
S s2 = { a = 10, b = 20 };
S s3 = { a = 5,  b = 20 };

// ✅ 正确：相同类型，可以比较
bool eq1 = (s1 == s2);    // eq1 = true（所有成员都相等）
bool eq2 = (s1 == s3);    // eq2 = false（a 不相等）
bool neq = (s1 != s3);    // neq = true（不相等）
```

------

**示例 2：嵌套结构体比较（递归比较）**

```cpp
struct Inner {
    bit<8> x;
    bit<8> y;
}

struct Outer {
    bit<16> id;
    Inner inner;
}

Outer o1 = { id = 100, inner = { x = 1, y = 2 } };
Outer o2 = { id = 100, inner = { x = 1, y = 2 } };
Outer o3 = { id = 100, inner = { x = 1, y = 3 } };

// ✅ 正确：所有成员递归比较
bool eq = (o1 == o2);    // eq = true（id 相等，inner.x 相等，inner.y 相等）
bool eq2 = (o1 == o3);   // eq2 = false（inner.y 不相等）
```

------

**示例 3：不同类型的结构体不能比较 ❌**

```cpp
struct S1 {
    bit<32> a;
    bit<32> b;
}

struct S2 {
    bit<32> a;
    bit<32> b;
}

S1 x = { a = 1, b = 2 };
S2 y = { a = 1, b = 2 };

// ❌ 错误：类型不同，不能比较
bool result = (x == y);    // 编译错误！
```

> **注意**：即使 `S1` 和 `S2` 的成员完全一样，它们也是**不同的类型**，不能直接比较！



### 3.3.3 `tuple`：元组类型核心要点总结

**1. 定义**

- 不可变的有序复合数据类型
- 只能**整体赋值**，不能单独修改成员

```cpp
tuple<bit<32>, bool> x = { 10, false };
```

**一、元组的定义和整体赋值（✅ 可行）**

```cpp
// 定义一个元组变量
tuple<bit<32>, bool> x = { 10, false };

// ✅ 整体赋值：相同类型的元组整体赋值
tuple<bit<32>, bool> y = { 20, true };
x = y;                    // x 整体变为 { 20, true }

// ✅ 整体赋值：直接赋值新的元组
x = { 30, false };        // x 整体变为 { 30, false }

// ✅ 作为返回值整体传递
tuple<bit<32>, bool> getTuple() {
    return { 100, true };
}
x = getTuple();           // x 整体变为 { 100, true }
```

------

**二、单独修改成员（❌ 不允许）**

```cpp
tuple<bit<32>, bool> x = { 10, false };

// ❌ 错误：元组的成员不是左值，不能单独修改
x[0] = 20;                // 编译错误！
x[1] = true;              // 编译错误！

// ❌ 错误：不能对成员进行赋值操作
x[0] = x[0] + 5;          // 编译错误！
```

------

**三、正确修改成员的方法（变通方式）**

如果想修改元组的某一个成员，只能**整体重新赋值**：

```cpp
tuple<bit<32>, bool> x = { 10, false };

// ✅ 方法1：整体赋值一个新的元组
x = { 20, false };        // 只修改第一个成员

// ✅ 方法2：先读取旧值，修改后再整体赋值
bit<32> old_val = x[0];   // ✅ 可以读取
x = { old_val + 5, x[1] }; // 整体赋值：x[0] 加 5，x[1] 不变

// ✅ 方法3：借助临时变量
tuple<bit<32>, bool> temp = x;  // 读取整体
temp = { 30, x[1] };            // 修改临时变量
x = temp;                       // 整体赋值回去
```

---

**2. 支持的操作**

**1. "元组整体可以作为左值"**

意思是：整个元组变量可以放在 `=` 左边，被整体赋值。

```python
tuple<bit<32>, bool> x = { 10, false };
tuple<bit<32>, bool> y = { 20, true };

x = y;        // ✅ 整个元组 x 作为左值，被整体赋值
x = { 30, false };  // ✅ 整个元组 x 作为左值
```

------

**2. "元组的成员不是左值"**

意思是：元组的单个成员 `x[0]`、`x[1]` **不能**放在 `=` 左边，不能被单独赋值。

```python
tuple<bit<32>, bool> x = { 10, false };

x[0] = 20;    // ❌ 错误！x[0] 不是左值，不能单独修改
x[1] = true;  // ❌ 错误！x[1] 不是左值，不能单独修改
```

------

**3. "一个元组可以整体赋值，但是元组的一个成员不能被单独修改"**

这就是上面两点的总结：

```python
x = { 20, true };    // ✅ 整体赋值，可行
x[0] = 20;           // ❌ 单独修改成员，不可行
```

------

**4. "这个限制可能在后续版本的 P4 语言规范中解除"**

意思是：现在的 P4 版本（1.2.3）有这个限制，但未来可能会放开，让元组成员也可以单独修改。

### 3.3.4 `header`：类型

**1. 定义**

header 是 P4 中最核心的类型，用于表示报文头部。

```cpp
header Ethernet_h {
    bit<48> dstAddr;
    bit<48> srcAddr;
    bit<16> etherType;
}
```

**header 类型与struct 结构类型相似，也可以包含多个字段。那为什么 P4 要单独设计一个 header 类型呢？**

header 类型额外包含一个隐藏的 validity 字段，该字段是布尔类型。当 validity 字段为 true 时，表示该 header 是有效的。在接收报文时，解析到的 header 会设置其 validity 字段为 true，没有解析到的 header 会保持其 validity 字段为 false。在发送报文时，有效 header 会被发送出去，无效的 header 不会被发送出去。使用 header 类型定义局部变量时，其 validity 位将自动设置为 false。基于 header 的 validity 字段，可以更方便地标识协议是否在，以及控制协议头的插入和删除。这些功能是 struct 类型不具备的。

**与 struct 的区别**

| 特性                    | struct   | header                          |
| :---------------------- | :------- | :------------------------------ |
| 隐藏的 validity 字段    | ❌ 无     | ✅ 有（布尔类型）                |
| 表示协议头是否存在      | ❌ 不能   | ✅ 能（valid 表示存在）          |
| 解析时自动设置 validity | ❌ 不支持 | ✅ extract() 成功时设为 true     |
| 发送时只发有效 header   | ❌ 不支持 | ✅ emit() 只发送 valid 的 header |
| header 嵌套             | N/A      | ❌ 不支持嵌套 header             |

> **总结：header = struct + validity 字段**



**2. header的性质**

**关键规则**

- header 类型定义局部变量时，**validity 位自动设为 false**
- header 类型**不支持嵌套**（不能 header 里套 header）
- 但 **struct 可以嵌套 header**
- 推荐用法：**header 定义单个协议头，struct 组合多个协议头**



下面的例子使用 header 定义了三个协议头：Ethernet_h、ipv4_h 和 ipv6_h，并使用 struct 将它们组合起来，构成一组报文头。当处理 IPv4 报文的时候，ipv4 头部的 validity 字段为 true，ipv6 头部的 validity 字段为 false。当处理 IPv6 报文的时候，ipv4 头部使用 validity 字段为 false，ipv6 头部的 validity 字段为 true。

**1. 为什么 IPv4 场景下 `ipv4.valid = true`，`ipv6.valid = false`？**

- 当收到一个 IPv4 报文时：
  - 以太网头部的 `etherType` 字段值为 `0x0800`。
  - 解析器根据这个值，**只调用解析 `ipv4_h` 的逻辑**，将报文内容填入 `ipv4` 字段，并自动将该头部的 valid 标记设为 `true`。
  - 因为解析器**没有执行**解析 `ipv6_h` 的代码，所以 `ipv6` 头部的 valid 保持初始状态 `false`（P4 中 header 默认是 invalid 的）。

------

**2. 为什么 IPv6 场景下 `ipv4.valid = false`，`ipv6.valid = true`？**

- 当收到一个 IPv6 报文时：
  - 以太网头部的 `etherType` 字段值为 `0x86DD`。
  - 解析器根据这个值，**只调用解析 `ipv6_h` 的逻辑**，将报文内容填入 `ipv6` 字段，并自动将该头部的 valid 标记设为 `true`。
  - 同样，因为没有解析 `ipv4_h`，所以 `ipv4` 的 valid 保持 `false`。

------

**3. 为什么不会两个同时为 true？**

- 因为一个报文在数据链路层之后，**只能是 IPv4 或 IPv6 之一**，不可能同时是两者。
- 你的 `struct headers_t` 虽然同时包含两个 IP 头部的定义，但这是为了**复用结构体**，在实际报文中，**同一时刻只有一个 IP 头部是有效的**。
- 解析器通过 `select` 语句根据 EtherType 做**互斥解析**，确保只设置其中一个为 valid。

---



```cpp
header Ethernet_h {
    bit<48> dstAddr;
    bit<48> srcAddr;
    bit<16> etherType;
}

header ipv4_h {
    bit<4> version;
    bit<4> ihl;
    bit<8> diffserv;
    bit<16> total_len;
    bit<16> identification;
    bit<3> flag;
    bit<13> flag_offset;
    bit<8> ttl;
    bit<8> protocol;

```

```cpp
struct ipv6_addr_t {
```

**3. 固定长度 vs 可变长度 Header**

**固定长度 Header**

- **所有字段都是固定长度**（如 `bit<8>`、`bit<16>`、`ipv4_addr_t` 等）。
- **总长度是常量**，在编译时就能确定。
- **validity 位不计入长度**（validity 是每个 header 实例的隐式状态位，不占用报文空间）。

```cpp
// 固定长度示例：长度 = 48+48+16 = 112 bit = 14 字节
header Ethernet_h {
    bit<48> dstAddr;
    bit<48> srcAddr;
    bit<16> etherType;
}
```

**可变长度 Header**

- **包含至少一个 `varbit` 字段**。
- 总长度 = 固定部分长度 + `varbit` 字段的实际长度（在运行时确定）。
- 常用于 IPv4 选项、TCP 选项等场景。

```cpp
// 可变长度示例：固定部分 160bit + options 动态长度
header ipv4_h1 {
    bit<4>  version;
    bit<4>  ihl;
    bit<8>  diffserv;
    bit<16> total_len;
    bit<16> identification;
    bit<3>  flags;
    bit<13> frag_offset;
    bit<8>  ttl;
    bit<8>  protocol;
    bit<16> hdr_checksum;
    bit<32> src_addr;
    bit<32> dst_addr;
    varbit<320> options;   // 最大 320 bit，实际长度由 ihl 字段决定
}
```

------

**`varbit` 字段的关键特性**

**语法**

```cpp
varbit<最大位宽> 字段名;
```

- `<最大位宽>` 是一个**常量表达式**，指定该字段能占用的最大位数。
- **实际长度**在运行时由解析器（Parser）根据报文字段（如 `ihl`）动态决定。

**长度计算规则**

- **Header 固定部分长度** = 所有非 `varbit` 字段的位宽之和。
- **Header 总长度** = 固定部分长度 + `varbit` 字段实际长度（由解析器传入）。
- **注意**：`varbit` 字段**必须放在 header 的最后**，不能中间穿插其他字段。

**为什么有最大位宽限制？**

- 编译器需要为 `varbit` 字段预留存储空间上限，以便在控制平面进行内存分配。
- 实际解析时，可以提取少于最大值的位数。

------

**IPv4 可变长度 Header 的典型应用**

**场景：解析 IPv4 选项字段**

IPv4 头部中的 `IHL`（Internet Header Length）字段指示整个 IPv4 头部的长度（以 32 位字为单位）。选项字段的长度 = `(IHL - 5) * 32` 位。

**解析器中的实现**

```cpp
parser IPv4Parser(packet_in pkt, out headers_t hdr) {
    state parse_ipv4 {
        pkt.extract(hdr.ipv4);  // 先提取固定部分（包括 version, ihl 等）
        transition select(hdr.ipv4.ihl) {
            5: accept;           // 没有选项字段
            _: parse_ipv4_options; // 有选项字段
        }
    }
    state parse_ipv4_options {
        // 计算选项字段长度：(ihl - 5) * 32 bit
        bit<16> opt_len = (bit<16>)(hdr.ipv4.ihl - 5) * 32;
        pkt.extract(hdr.ipv4.options, opt_len);  // 动态指定提取长度
        transition accept;
    }
}
```

**关键点**

- `pkt.extract()` 的第二个参数用于**动态指定 `varbit` 字段的实际长度**。
- 提取完成后，`hdr.ipv4.options` 在控制平面中可通过 `hdr.ipv4.options.size()` 获取实际长度。

------

**Target 相关的限制**

**字节对齐限制**

- 虽然 P4 规范**不强制要求** header 总长度为 8 的倍数，但大多数 **Target（如 Tofino、v1model）要求 header 总长度为 8 的整数倍**（即字节对齐）。
- 如果固定部分 + `varbit` 最大长度不是 8 的倍数，编译器可能报错或自动填充。

**示例：不满足字节对齐的情况**

```cpp
header BadHeader {
    bit<3>  field1;      // 3 bit
    varbit<5> options;   // 最大 5 bit
    // 总长度最大为 8 bit，满足字节对齐，OK
}

header BadHeader2 {
    bit<3>  field1;      // 3 bit
    varbit<6> options;   // 最大 6 bit
    // 总长度最大为 9 bit，不是 8 的倍数，可能在某些 Target 上报错
}
```





**4. header支持的操作**

**validity 操作方法**

| 方法           | 作用                        |
| :------------- | :-------------------------- |
| `isValid()`    | 判断 header 是否有效        |
| `setValid()`   | 设置 header 为有效（true）  |
| `setInvalid()` | 设置 header 为无效（false） |

```cpp
if (hdr.eth.isValid()) {
    // 以太网头部存在
}

hdr.ipv4.setValid();    // 手动设为有效
hdr.ipv6.setInvalid();  // 手动设为无效
```

**重要规则**

> **当 header 为无效时，对该 header 的修改不生效！必须先 `setValid()` 再修改！**

```cpp
H h;                    // validity 自动为 false
h.x = 10;               // ❌ 修改不生效（无效 header）

h.setValid();           // ✅ 先设为有效
h.x = 10;               // ✅ 修改生效
```

------

**extract() 操作**

- 解析器（parser）中用于从报文填充 header
- 执行成功后，**validity 位自动设为 true**

```cpp
packet.extract(hdr.eth);    // 解析以太网头，自动设 valid
```

------

**emit() 操作**

- 发送器（deparser）中用于发送 header
- **只发送有效的 header**

```cpp
packet.emit(hdr.eth);       // 只有 valid 时才发送
```

------

**header 初始化**

- 使用列表表达式或 structure-valued 表达式
- **初始化时 validity 位自动设为 true**

```cpp
H h1 = { 10, 12 };           // valid = true
H h2 = { y = 12, x = 10 };   // valid = true
```

------

**header 比较**

- 相同类型 header 支持 `==` 和 `!=`
- **比较条件**：validity 位相同 **且** 所有对应成员值相同

```cpp
if (h1 == h2) {    // 同时比较 validity 和所有字段
    // 完全相等
}
```

### 3.3.5 `header stack`：类型

**1. 定义**

多个**相同类型** header 组成的**数组**。

```cpp
header Mpls_h {
    bit<20> label;
    bit<3> tc;
    bit bos;        // bottom of stack，栈底标志
    bit<8> ttl;
}

struct Pkthdr {
    Ethernet_h ethernet;
    Mpls_h[3] mpls;    // 3 个 MPLS 标签组成的数组
}
```

------

**2. 核心机制：next 和 last 指针**

| 指针        | 含义                                       |
| :---------- | :----------------------------------------- |
| `mpls.next` | 指向**下一个**需要被解析的元素（初始为 0） |
| `mpls.last` | 指向**已经被解析的最后一个**元素           |

- 每次 `extract()` 成功后，`mpls.next` 自动加 1
- 使用这两个指针可以轻松实现**循环解析**

```cpp
header Mpls_h {
    bit<20> label;
    bit<3> tc;
    bit bos;
    bit<8> ttl;
}

struct Pkthdr {
    Ethernet_h ethernet;
    Mpls_h[3] mpls;
    // 其他头部省略
}

parser P(packet_in b, out Pkthdr p) {
    state start {
    b.extract(p.ethernet);
    transition select(p.ethernet.etherType);
    }
    0x8847: parse_mpls;
    0x0800: parse_ipv4;

    state parse_mpls {
    b.extract(p.mpls.next);
    transition select(p.mpls.last.bos);
    	 0: parse_mpls;          // bos=0，继续解析
         1: parse_ipv4;          // bos=1，栈底，去解析 IPv4
    }
}

// 第二种写法
parser P(packet_in b, out Pkthdr p) {
    state start {
   		b.extract(p.ethernet);
        transition select(p.ethernet.etherType) {
            0x8847: parse_mpls;
            0x0800: parse_ipv4;
        }
        
    }

    state parse_mpls {
        b.extract(p.mpls.next);
        transition select(p.mpls.last.bos) {
             0: parse_mpls;          // bos=0，继续解析
             1: parse_ipv4;          // bos=1，栈底，去解析 IPv4
        }
    }
}
```



**3. 代码解析**

**第一步：先搞清楚 MPLS 是什么**

MPLS 是一种网络协议，它的特点是**可以有多层标签（像千层饼一样叠在一起）**：

```text
报文结构：
[以太网头] [MPLS标签1] [MPLS标签2] [MPLS标签3] [IP头] [数据]
              ↑            ↑            ↑
            第1层        第2层        第3层（最后一层）
```

每一层 MPLS 标签中都有一个 **`bos` 字段**（bottom of stack，栈底标志）：

| bos 值    | 含义                                   |
| :-------- | :------------------------------------- |
| `bos = 0` | 后面还有 MPLS 标签，继续往下读         |
| `bos = 1` | 这是最后一层 MPLS 标签，后面是 IP 头了 |

------

**第二步：代码在做什么**

```cpp
header Mpls_h {
    bit<20> label;
    bit<3> tc;
    bit bos;      // 这个字段决定是否继续解析
    bit<8> ttl;
}
```

定义了一个 MPLS 标签头的格式，其中 **`bos` 字段就是"开关"**。

```cpp
struct Pkthdr {
    Ethernet_h ethernet;
    Mpls_h[3] mpls;    // 最多支持 3 层 MPLS 标签
}
```

定义了一个"大盒子"，里面最多放 3 个 MPLS 标签。

------

**第三步：parser 解析流程**

**刚开始解析时：**

```text
mpls.next = 0  ← 指向数组第 0 个位置
mpls.last = 未定义（还没有解析任何元素）
```

```cpp
state parse_mpls {
    b.extract(p.mpls.next);    // 解析到 mpls[0]
    transition select(p.mpls.last.bos) {
        // 判断刚解析完的那个标签的 bos 字段
    }
}
```

------

**第四步：图解完整执行过程**

**场景1：只有 1 层 MPLS 标签**

```text
报文：[以太网头] [MPLS标签(bos=1)] [IP头] [数据]

状态：parse_mpls（第一次进入）
  ↓
b.extract(p.mpls.next)
  → mpls.next 当前是 0，所以解析到 mpls[0]
  → 解析成功后，mpls.next 自动变成 1
  → mpls.last 变成 0（指向刚刚解析完的 mpls[0]）
  ↓
判断 p.mpls.last.bos
  → p.mpls[0].bos == 1  （因为这是最后一层）
  ↓
跳转到 parse_ipv4 （不再循环）
```

------

**场景2：有 3 层 MPLS 标签**

```cpp
报文：[以太网头] [标签1(bos=0)] [标签2(bos=0)] [标签3(bos=1)] [IP头] [数据]
              ↑                ↑                ↑
            mpls[0]          mpls[1]          mpls[2]
```

**第 1 次进入 `parse_mpls`：**

```
mpls.next = 0
  ↓
b.extract(p.mpls.next) → 解析到 mpls[0]
  ↓
mpls.next 自动变成 1
mpls.last = 0
  ↓
判断 p.mpls.last.bos → mpls[0].bos == 0
  ↓
跳转到 parse_mpls（循环！继续解析）
```

**第 2 次进入 `parse_mpls`：**

```
mpls.next = 1
  ↓
b.extract(p.mpls.next) → 解析到 mpls[1]
  ↓
mpls.next 自动变成 2
mpls.last = 1
  ↓
判断 p.mpls.last.bos → mpls[1].bos == 0
  ↓
跳转到 parse_mpls（循环！继续解析）
```

**第 3 次进入 `parse_mpls`：**

```
mpls.next = 2
  ↓
b.extract(p.mpls.next) → 解析到 mpls[2]
  ↓
mpls.next 自动变成 3
mpls.last = 2
  ↓
判断 p.mpls.last.bos → mpls[2].bos == 1
  ↓
跳转到 parse_ipv4（停止循环，去解析 IP 头）
```

------

**第五步：动画演示指针变化**

```text
初始状态：
mpls 数组: [ 空 ] [ 空 ] [ 空 ]
mpls.next = 0  ← 指向这里
mpls.last = 无


第1次 extract 后：
mpls 数组: [标签1] [ 空 ] [ 空 ]
mpls.next = 1  ← 指向这里
mpls.last = 0  ← 指向这里（刚解析完）


第2次 extract 后：
mpls 数组: [标签1] [标签2] [ 空 ]
mpls.next = 2  ← 指向这里
mpls.last = 1  ← 指向这里（刚解析完）


第3次 extract 后：
mpls 数组: [标签1] [标签2] [标签3]
mpls.next = 3  ← 已经超出数组范围（停止）
mpls.last = 2  ← 指向这里（刚解析完）
```

------

**第六步：核心理解**

**`mpls.next` 的初始值为什么是 0？**

**因为数组下标从 0 开始！**

- `mpls[0]` = 数组第 1 个元素
- `mpls[1]` = 数组第 2 个元素
- `mpls[2]` = 数组第 3 个元素

**每次 `extract()` 成功后，`mpls.next` 为什么会自动加 1？**

**这就是 P4 编译器帮我们做的"自动指针移动"！**

我们不需要手动写 `mpls.next = mpls.next + 1`，编译器自动处理了。

**为什么要区分 `next` 和 `last`？**

| 指针        | 指向               | 用途                          |
| :---------- | :----------------- | :---------------------------- |
| `mpls.next` | 下一个要解析的位置 | 告诉 extract 往哪里填数据     |
| `mpls.last` | 刚解析完的那个元素 | 用来判断 bos 字段，决定下一步 |

------

**第七步：一句话总结**

> **`next` 告诉程序"下一个填哪里"，`last` 告诉程序"刚才填了什么"，两者配合实现循环解析多层 MPLS 标签。**

------

**完整流程图**

```text
                    ┌─────────────────┐
                    │   start 状态    │
                    │ 解析以太网头    │
                    └────────┬────────┘
                             │
                    select(etherType)
                    ┌────────┴────────┐
                    │                 │
              0x8847              0x0800
              (MPLS)              (IP)
                    │                 │
                    ▼                 ▼
              ┌───────────┐      parse_ipv4
              │parse_mpls │
              └─────┬─────┘
                    │
           b.extract(p.mpls.next)
                    │
              mpls.next 自动 +1
                    │
           select(p.mpls.last.bos)
              ┌─────┴─────┐
              │           │
           bos=0       bos=1
              │           │
              ▼           ▼
        返回 parse_mpls  parse_ipv4
        （继续循环）    （退出循环）
```

------



**4. 代码运行顺序**

**P4 程序的运行模型**

**P4 程序 = 硬件流水线配置**

P4 程序不是"运行"的，而是**告诉硬件（交换机/网卡）如何处理每个到达的数据包**。

```text
每个数据包到达时，硬件自动执行：
         ↓
    [Parser]      ← 解析报文头
         ↓
    [Match-Action] ← 查表、处理
         ↓
    [Deparser]    ← 组装并发送报文
```

------

**所以您看到的代码其实是"配置"，不是"主函数"**

**在您的代码中：**

```cpp
parser P(packet_in b, out Pkthdr p) {
    state start {
        b.extract(p.ethernet);
        transition select(p.ethernet.etherType);
    }
    // 注意：这行不在任何 state 里面！
    0x8847: parse_mpls;
    0x0800: parse_ipv4;

    state parse_mpls {
        b.extract(p.mpls.next);
        transition select(p.mpls.last.bos);
    }
}
```

**这段代码的真实含义：**

**不是按顺序执行的！** 而是定义了**状态机**：

```
每个数据包从 "start" 状态开始
         ↓
    解析以太网头
         ↓
    根据 etherType 跳转
         ↓
    ┌────┴────┐
    │         │
0x8847    0x0800
(MPLS)    (IPv4)
    │         │
    ▼         ▼
parse_mpls  parse_ipv4
    │
    ▼
判断 bos 字段
```

------

**对比理解**

**C 语言（顺序执行）：**

```cpp
int main() {
    step1();  // 先执行
    step2();  // 再执行
    step3();  // 最后执行
}
```

**P4（状态机跳转）：**

```cpp
parser P {
    state start {
        // 执行完这里
        transition 跳转到某个状态;  // 不是执行下一行！
    }
    // 这里的代码不会自动执行，需要被跳转过来
    state parse_mpls {
        // 执行完这里
        transition 跳转到下一个状态;
    }
}
```

------

**具体到您的代码，执行顺序是：**

**每个数据包到达时的实际流程：**

```text
┌─────────────────────────────────────────────────┐
│  数据包到达（例如：一个带 3 层 MPLS 的报文）     │
└─────────────────────────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────┐
│  自动进入 parser 的 "start" 状态                │
│  → 执行: b.extract(p.ethernet)                 │
│  → 执行: transition select(etherType)          │
│  → 发现 etherType == 0x8847 (MPLS)            │
│  → 跳转到 "parse_mpls"                         │
└─────────────────────────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────┐
│  第 1 次进入 "parse_mpls" 状态                  │
│  → 执行: b.extract(p.mpls.next)  // 解析第1个标签│
│  → 执行: transition select(bos)               │
│  → 发现 bos == 0 (后面还有标签)                │
│  → 跳转到 "parse_mpls"（循环！）               │
└─────────────────────────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────┐
│  第 2 次进入 "parse_mpls" 状态                  │
│  → 执行: b.extract(p.mpls.next)  // 解析第2个标签│
│  → 执行: transition select(bos)               │
│  → 发现 bos == 0 (后面还有标签)                │
│  → 跳转到 "parse_mpls"（循环！）               │
└─────────────────────────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────┐
│  第 3 次进入 "parse_mpls" 状态                  │
│  → 执行: b.extract(p.mpls.next)  // 解析第3个标签│
│  → 执行: transition select(bos)               │
│  → 发现 bos == 1 (这是最后一层)                │
│  → 跳转到 "parse_ipv4"                         │
└─────────────────────────────────────────────────┘
                      ↓
             继续处理 IPv4 头...
```

------

**关键理解**

**1. 没有 main() 函数**

P4 程序的"入口"是 **parser 的 `start` 状态**，每个数据包都从这里开始。

**2. 不是顺序执行，而是状态机跳转**

- C 语言：执行完第 1 行，自动执行第 2 行
- P4 语言：执行完当前状态，通过 `transition` **显式跳转**到下一个状态

**3. 相同的状态可以被多次进入**

在上面的例子中，`parse_mpls` 被进入了 3 次（因为 MPLS 有 3 层标签）

**4. 代码中的 `0x8847: parse_mpls;` 不是"函数调用"**

这是**状态跳转规则**，意思是："如果 etherType 是 0x8847，就跳转到 parse_mpls 状态"







### 3.3.6 `header union`：类型

**1. 定义**

类似于 C 语言的 `union`，多个不同的 header 类型可以组合成一个 union，**共享底层存储空间**，同一时刻只有一个成员有效。，不同的成员共享底层的存储空间，同一个时刻只有一个成员是有效的。一个 header union 中的每个成员都必须是 header 类型。

```cpp
header_union IP_h {
    IPv4_h v4;
    IPv6_h v6;
}

struct Parsed_packet {
    Ethernet_h ethernet;
    IP_h ip;          // 要么是 IPv4，要么是 IPv6
}
```

------

**2. 特点**

| 特性                          | 说明                           |
| :---------------------------- | :----------------------------- |
| 成员必须是 header 类型        | ✅                              |
| 共享存储空间                  | ✅                              |
| 同一时刻只有一个有效          | ✅                              |
| `isValid()`                   | ✅ 至少一个成员有效时返回 true  |
| `setValid()` / `setInvalid()` | ❌ 不支持                       |
| `==` / `!=`                   | ✅ 比较validity位和对应的成员值 |

------

**3. 典型应用场景**

IP 报文要么是 IPv4 要么是 IPv6，用 header union 表示非常合适。

```cpp
// 解析时根据 etherType 决定解析 IPv4 还是 IPv6
transition select(p.ethernet.etherType) {
    0x0800 : parse_ipv4;    // IPv4
    0x86DD : parse_ipv6;    // IPv6
}

state parse_ipv4 {
    b.extract(p.ip.v4);     // 填充 IPv4
    transition accept;
}

state parse_ipv6 {
    b.extract(p.ip.v6);     // 填充 IPv6
    transition accept;
}
```

------

### 3.3.7 集合类型

**1. 定义**

表示一组常量的集合，主要用于 parser 的 `select` 语句中。

```cpp
select (expression) {
    set1: state1;
    set2: state2;
    default: default_state;
}
```

set1和set2代表两个数值集合，select表达式用于判断expression所表示的值是否属于其中某个集合

集合类型包含两种特殊情况

------

**2. 两种特殊情况**

| 类型   | 写法              | 含义                |
| :----- | :---------------- | :------------------ |
| 单个值 | `4: continue;`    | 只匹配值为 4 的情况 |
| 全集   | `default:` 或 `_` | 匹配所有情况        |

```cpp
select (hdr.ipv4.version) {
    4: continue;         // 只匹配 version = 4
    default: reject;     // 其他全部拒绝
}
```

------

下面的程序代码中，下面线（_）表示 hdr.ipv4.version 所有可能取值的全集：

```cpp
select (hdr.ipv4.version) {
   4: continue;
   _: reject;
}
```

**注意：**集合类型是 P4 编译器自动合成的类型，程序员无法直接定义该类型的变量。P4 中并没有定义集合类型 set 关键字。集合类型只能在 P4 的少数场景中使用。



### 快速对比总结表

| 类型             | 特点                                  | 主要用途                         |
| :--------------- | :------------------------------------ | :------------------------------- |
| **header**       | 有 validity 位，表示协议头是否存在    | 定义单个协议头                   |
| **struct**       | 无 validity 位，可嵌套 header         | 组合多个 header                  |
| **header stack** | 相同 header 的数组，有 next/last 指针 | 处理重复协议头（如 MPLS 标签栈） |
| **header union** | 共享存储，同一时刻只有一个有效        | 表示互斥的协议（如 IPv4/IPv6）   |
| **集合类型**     | 常量的集合                            | parser 的 select 匹配            |



### 3.3.8 `extern`：类型

**核心概念**

- **extern**：P4 Target 提供的**外部功能组件**
- 代码和实现**不公开**，通过对象和方法**暴露接口**
- 在特定 Target 的编程手册中提供详细描述

**常见组件**

| 组件            | 用途                   |
| :-------------- | :--------------------- |
| `Hash`          | 计算哈希值（如 CRC16） |
| `Checksum16`    | 增量计算 16 位校验和   |
| `ActionProfile` | 动作配置文件           |
| `register`      | 寄存器数组             |

**代码示例**

**Hash 组件（BMv2）：**

```cpp
Hash<bit<16>>(HashAlgorithm_t.CRC16) ip_hash_rand;
apply {
    eg_md.inner_ip_hash_value = ip_hash_rand.get({
        hdr.inner_ipv4.src_addr,
        hdr.inner_ipv4.dst_addr
    });
}
```

**Checksum16 组件：**

```cpp
extern Checksum16 {
    Checksum16();                // 构造函数
    void clear();                // 清零校验和
    void update<T>(in T data);   // 增量更新校验和
}
```



## 3.4 有关数据类型的其他主题

### 3.4.1 类型默认值

| 类型                      | 默认值                     |
| :------------------------ | :------------------------- |
| `int`、`bit<W>`、`int<W>` | `0`                        |
| `bool`                    | `false`                    |
| `error`                   | `error.NoError`            |
| `varbit<W>`               | 动态位宽为 0 的位串        |
| 有底层类型的 `enum`       | `0`（即使不是合法标签）    |
| 无底层类型的 `enum`       | 声明中的**第一个值**       |
| `header`                  | `invalid`                  |
| `header stack`            | 所有元素为 `invalid`       |
| `header union`            | 所有元素为 `invalid`       |
| `struct`                  | 每个字段为对应类型的默认值 |
| `tuple`                   | 每个字段为对应类型的默认值 |



### 3.4.2 未初始化的值

**核心规则**

- **是否自动初始化取决于编译器/Target**
- 读取未初始化的变量 → **未定义行为**

**危险场景**

1. 从 `invalid` 的 header 读取成员
2. 从 `valid` 但**未赋值**的 header 读取成员
3. 读取未初始化的 struct 成员

**防范措施**

- 尽量**显式初始化**所有变量
- `enum`/`error` 未初始化时可能不等于任何标签值
- `select` 中必须添加 **`default` 或 `_` 分支**处理未定义值



### 3.4.3 类型转换

**显式类型转换**

| 转换                           | 规则                              |
| :----------------------------- | :-------------------------------- |
| `bit<1>` ↔ `bool`              | `0↔false`，`1↔true`               |
| `bit<W>` → `bit<X>`            | W > X：高位截断；否则高位补 0     |
| `int<W>` → `bit<W>`            | 保持所有位不变，负值转正值        |
| `bit<W>` → `int<W>`            | 保持所有位不变，最高位 1 时转负值 |
| `int<W>` → `int<X>`            | W > X：高位截断；否则符号位展开   |
| 有底层类型的 `enum` ↔ 底层类型 | 可以强制转换                      |

**隐式类型转换**

> **P4 只支持以下两种隐式转换：**

| 转换                        | 说明                   |
| :-------------------------- | :--------------------- |
| `int` → `bit<W>` / `int<W>` | 整数自动转为指定位宽   |
| `enum` → 底层类型           | 枚举自动转为其底层类型 |

**隐式转换示例**

```cpp
x + 1      → x + (bit<8>)1      // bit<8> + int → 都转 bit<8>
x | 0xFFF  → x | (bit<8>)0xFFF  // 溢出警告
x + E.a    → x + (bit<8>)E.a    // enum 自动转底层类型
z < 0      → z < (int<8>)0      // int 自动转 int<8>
```



### 3.4.4 `typedef`：类型别名

```cpp
typedef bit<32> u32;          // 32位无符号整数
typedef bit<48> mac_addr_t;   // MAC地址
typedef bit<128> ipv6_addr_t; // IPv6地址
```

> **原始类型和新类型被视为同义词**，新类型支持原始类型的所有操作。



### 3.4.5 类型嵌套规则

| 元素类型       | header | header union | struct/tuple |
| :------------- | :----- | :----------- | :----------- |
| `bit<W>`       | ✅      | ❌            | ✅            |
| `int<W>`       | ✅      | ❌            | ✅            |
| `varbit<W>`    | ✅      | ❌            | ❌            |
| `int`          | ❌      | ❌            | ✅            |
| `void`         | ❌      | ❌            | ✅            |
| `error`        | ❌      | ❌            | ✅            |
| `bool`         | ✅      | ❌            | ✅            |
| `enum`         | ✅      | ❌            | ✅            |
| `header`       | ❌      | ❌            | ✅            |
| `header stack` | ❌      | ❌            | ✅            |
| `header union` | ✅      | ❌            | ✅            |
| `struct`       | ✅      | ❌            | ✅            |
| `tuple`        | ❌      | ❌            | ✅            |

**关键结论**

> **`int` 类型不能在 header/header union 中使用**（没有明确的存储空间大小）
>
> **`varbit<W>` 只能在 header 中使用**



### 3.4.6 运算符优先级

**P4 vs C 语言差异**

| 差异点            | 说明                                                     |
| :---------------- | :------------------------------------------------------- |
| 位运算符优先级    | P4 中 `&`、`|`、`^` **高于**关系运算符（C 中相反）       |
| `++` 运算符       | P4 中是**拼接运算符**（不是自增）                        |
| 切片 `[m:n]`      | 与索引 `[i]` 优先级相同                                  |
| P4 不支持的运算符 | `->`、`&`（取地址）、`sizeof`、`++`、`--`、`*=`、`/=` 等 |



**表3-4 C语言运算符优先级**

| 运算符                              | 优先级 | 说明           |
| :---------------------------------- | :----- | :------------- |
| `0 () ->`                           | 1      | 括号、成员访问 |
| `! ~ + - * & (type) sizeof`         | 2      | 一元运算符     |
| `* / %`                             | 3      | 乘除取模       |
| `+ -`                               | 4      | 加减           |
| `<< >>`                             | 5      | 移位           |
| `< <= >=`                           | 6      | 关系运算符     |
| `== !=`                             | 7      | 相等比较       |
| `&`                                 | 8      | 按位与         |
| `^`                                 | 9      | 按位异或       |
| `|`                                 | 10     | 按位或         |
| `&&`                                | 11     | 逻辑与         |
| `||`                                | 12     | 逻辑或         |
| `?:`                                | 13     | 条件运算符     |
| `= += -= *= /= %= &= ^= |= <<= >>=` | 14     | 赋值运算符     |
| `,`                                 | 15     | 逗号运算符     |



**P4 运算符优先级（表 3-5）**

| 优先级 | 运算符                   | 说明       |
| :----- | :----------------------- | :--------- |
| 1      | `()` `[]` `[m:n]`        | 索引、切片 |
| 2      | `!` `~` `+` `-` `(type)` | 一元运算符 |
| 3      | `*` `/` `%`              | 乘除取模   |
| 4      | `+` `-` `++`             | 加减、拼接 |
| 5      | `<<` `>>`                | 移位       |
| 6      | `&`                      | 按位与     |
| 7      | `^`                      | 按位异或   |
| 8      | `|`                      | 按位或     |
| 9      | `<` `<=` `>` `>=`        | 关系运算符 |
| 10     | `==` `!=`                | 相等比较   |
| 11     | `&&`                     | 逻辑与     |
| 12     | `||`                     | 逻辑或     |
| 13     | `?:`                     | 条件运算符 |
| 14     | `=`                      | 赋值       |



### 3.4.7 表达式的求值顺序

- 复合语句按**从左到右**顺序求值
- `&&`、`||` 支持**短路求值**
  - `a || b`：a 为 true 时不求值 b
  - `a && b`：a 为 false 时不求值 b



### 3.4.8 非法算术表达式

假设 `bit<8> x; bit<16> y; int<8> z;`

| 非法表达式  | 原因                   | 正确写法                                 |
| :---------- | :--------------------- | :--------------------------------------- |
| `x + y`     | 位宽不同               | `(bit<16>)x + y` 或 `x + (bit<8>)y`      |
| `x + z`     | 符号不同               | `(int<8>)x + z` 或 `x + (bit<8>)z`       |
| `(int<8>)y` | 不能同时改变符号和位宽 | `(int<8>)(bit<8>)y` 或 `(int<16>)y` 再转 |
| `y + z`     | 位宽和符号都不同       | 多种强转方式                             |
| `x << z`    | 右操作数不能是 `int`   | `x << (bit<8>)z`                         |
| `1 << x`    | `1` 的位宽未知         | `32w1 << x`                              |
| `~1`        | `int` 不支持位操作     | `~32w1`                                  |
| `5 & ~3`    | `int` 不支持位操作     | `32w5 & ~3`                              |



## 3.5 函数

**基本概念**

P4 中的函数和其他大部分语言的函数类似，但是 **P4 不支持递归函数**。

下面的代码定义了一个函数，返回两个 32 位无符号整型的最大值：

```cpp
bit<32> max(in bit<32> left, in bit<32> right, out bit<32> max_value)
{
    max_value = right;
    if (left > right) {
        max_value = left;
    }
    return max_value;
}
```

------

**函数参数方向**

**P4 函数的所有的参数必须指定方向**，这是与其他语言显著的不同点。

| 方向     | 关键字  | 能否读   | 能否写   |
| :------- | :------ | :------- | :------- |
| 输入     | `in`    | ✅ 可以   | ❌ 不可以 |
| 输出     | `out`   | ❌ 不可以 | ✅ 可以   |
| 输入输出 | `inout` | ✅ 可以   | ✅ 可以   |

**为什么要引入"方向"？**

1. 符合芯片逻辑的特点（硬件模块的参数是有方向的）
2. 通过多个输出参数可以实现**多个返回值**的效果（P4 中没有指针）

**局部变量**

- 可以在函数中定义局部变量
- 作用域：从变量定义开始到函数结束

**return 语句**

- 使用 `return` 指定返回值
- 返回值为 `void` 类型的函数可以不指定 return 的参数

**"忽略"类型**

如果程序中使用不到函数的返回值，可以**丢弃**，使用下画线 `_`：

```cpp
bit<32> left = 32w123;
bit<32> right = 32w456;
bit<32> max_value = 0;
_ = max(left, right, max_value);    // 返回值被丢弃
```

------

**构造函数**

- 用于初始化由具体 Target 提供的 `extern` 对象
- 构造函数调用完成后返回一个对应类型的对象
- **编译时求值**，所有参数的值也必须编译时确定

**示例：**

```cpp
extern ActionProfile {
    ActionProfile(bit<32> size); // 构造函数
}

table tbl {
    action = { /* 代码省略 */ }
    implementation = ActionProfile(1024);
}
```

## 3.6 语句

**支持的语句类型**

| 语句类型 | P4 是否支持                                                  |
| :------- | :----------------------------------------------------------- |
| 顺序语句 | ✅ 支持                                                       |
| 分支语句 | ✅ 支持                                                       |
| 循环语句 | ❌ **不支持**（无 `for`、`while`、`do while`，也无 `break`、`continue`） |

------

### 3.6.1 赋值语句

- 使用等号（`=`）表示
- 执行顺序：先求值左边（产生左值），再求值右边，最后赋值
- 复合数据类型进行**递归赋值**

------

### 3.6.2 条件语句

**与 C 语言基本相同，但有一个关键区别：**

> **条件表达式必须是布尔表达式，不可以是整数，不会进行隐式类型转换！**

❌ **错误示例：**

```cpp
bit<4> i = 2;

if (i) {        // ❌ 错误！i 是整数，不是布尔值
    do_a();
} else {
    do_b();
}
```

✅ **正确写法：**

```cpp
bit<4> i = 2;

if (i == 2) {   // ✅ 正确！i == 2 是布尔表达式
    do_a();
} else {
    do_b();
}
```

**注意：条件语句不能在 parser 中使用！** parser 中需要使用 `select` 语句实现分支判断。

------

### 3.6.3 switch 语句

> **switch 语句只能在 control 结构中使用。**

**1. 带有 action_run 表达式的 switch 语句**

- 用于根据**匹配-动作表**的执行结果进一步处理
- 格式必须是：`t.apply().action_run`（`t` 是表名）
- 所有 switch 标签必须是表 `t` 的操作名称，或者是 `default`

```cpp
switch (t.apply().action_run) {
    action1: // 继续执行 action2:
    action2: { /* 代码省略 */ }
    // 执行完 action2 后不再执行 action3
    action3: { /* 代码省略 */ }
    default: { /* 代码省略 */ }
}
```

**与 C 语言的 switch 不同：**

| C 语言                    | P4 语言                      |
| :------------------------ | :--------------------------- |
| 需要显式 `break` 防止跳转 | **花括号 `{}` 控制是否跳转** |
| 无 `{}` 则继续执行        | 无 `{}` 则继续执行           |
| 有 `break` 则停止         | 有 `{}` 则停止               |

**关键规则：**

- switch 标签后面**没有 `{}`** → 自动跳转到下一个标签
- switch 标签后面**有 `{}`** → 不会自动转到下一个标签
- `default` 标签**与表项是否命中无关**（不代表未命中）

------

**2. 使用整型或枚举类型表达式的 switch 语句**

表达式结果必须为以下类型之一：

- `bit<W>`
- `int<W>`
- `enum`
- `error`

所有标签必须是**常量表达式**，且必须可以隐式转换为 switch 表达式的类型。

```cpp
switch (hdr.ethernet.etherType) {
    0x86dd: { /* IPv6 */ }
    0x0800: // 继续执行下一段代码
    0x0802: { /* 代码省略 */ }
    0xcafe: { /* 代码省略 */ }
    default: { /* 代码省略 */ }
}
```

**3. switch 语句注意事项：**

| 规则             | 说明                       |
| :--------------- | :------------------------- |
| 标签不能重复     | 两个标签相等 = 编译时错误  |
| 标签不需要全覆盖 | 不必包括所有可能值         |
| `default` 可选   | 如果存在，必须放在**最后** |

**4. 执行逻辑：**

1. 计算 switch 表达式
2. 匹配标签：
   - 匹配成功 → 执行对应语句
   - 不匹配且有 `default` → 执行 `default`
   - 不匹配且无 `default` → **不执行任何语句**（但表达式结果仍保持）

------

### 3.6.4 return 语句

- 用于从包含它的**函数、action 或 control** 中立即终止执行
- parser 中**不允许**使用 `return`

| 规则           | 说明                                       |
| :------------- | :----------------------------------------- |
| 有返回值的函数 | `return` 后必须跟表达式，类型必须匹配      |
| 执行后         | `out` / `inout` 参数的修改结果**仍然保持** |

------

### 3.6.5 exit 语句

- 立即终止**当前执行的所有块**的执行
- 从当前 action/control 及其所有调用方中退出

| 规则      | 说明                                       |
| :-------- | :----------------------------------------- |
| parser 中 | ❌ 不允许使用                               |
| 函数中    | ❌ 不允许使用                               |
| 执行后    | `out` / `inout` 参数的修改结果**仍然保持** |

------



## 3.6 语句 vs parser 使用限制总结

| 语句          | parser 中能否使用               |
| :------------ | :------------------------------ |
| `if` 条件语句 | ❌ **不能**（用 `select` 代替）  |
| `switch` 语句 | ❌ **不能**（只能在 control 中） |
| `return` 语句 | ❌ **不能**                      |
| `exit` 语句   | ❌ **不能**                      |
| `select` 语句 | ✅ **能**（parser 专用）         |



## 3.7 `control`

### 核心概念

P4 程序主要通过**匹配-动作表**对报文进行操作：

| P4 元素   | 类比                                 |
| :-------- | :----------------------------------- |
| `table`   | 类中的**数据部分**（匹配条件、动作） |
| `action`  | 类中的**方法**                       |
| `control` | **类名**（封装 table 和 action）     |

- control 用于封装匹配-动作表
- 第 2 章中的 `MyIngress` 就是一个 `control`

------



### 完整 `control` 示例

```cpp
control MyIngress(inout headers_t hdr,
                  inout standard_metadata_t standard_metadata) {
    
    // 4.1 定义 action（做什么）
    action Set_dstAddr() {
        hdr.eth.dstAddr = 0xaabbccddee02;            // 改目的MAC
        standard_metadata.egress_spec = 0x2;          // 从端口2发出
    }

    action NoAction() { }

    // 4.2 定义 table（什么条件做什么）
    table mac_match_tbl {
        key = {
            hdr.eth.dstAddr : exact;                 // 按目的MAC精确匹配
        }
        actions = {
            Set_dstAddr;
            NoAction;
        }
        const entries = {                            // 常量表项（固定规则）
            (0x112233445566) : Set_dstAddr();        // 匹配这个MAC → 执行Set_dstAddr
        }
        default_action = NoAction();                 // 未命中 → 什么都不做
    }

    // 4.3 apply —— 执行查表
    apply {
        // 方式1：只调用，不关心结果（原代码）
        mac_match_tbl.apply();

        // 方式2：判断是否命中（新加）
        if (mac_match_tbl.apply().hit) {
            // 命中时的处理
        } else {
            // 未命中时的处理
        }

        // 方式3：判断执行了哪个action（新加）
        switch (mac_match_tbl.apply().action_run) {
            Set_dstAddr: {
                // 执行了 Set_dstAddr
            }
            NoAction: {
                // 执行了 NoAction
            }
            default: {
                // 其他情况
            }
    	}
	}
}
```



### `control MyIngress` 代码逐行解析

**一、这个 control 是干什么的？**

**一句话：每个数据包进入交换机时，都会自动执行 `apply` 里面的代码。**

```cpp
apply {
    mac_match_tbl.apply();    // 就是这一行
}
```

所以**每个数据包**都会执行 `mac_match_tbl.apply()`。

------

**二、`mac_match_tbl.apply()` 执行时发生了什么？**

**第一步：拿什么去查？**

```cpp
key = {
    hdr.eth.dstAddr : exact;
}
```

**从当前数据包中取出目的MAC地址**（`hdr.eth.dstAddr`），拿这个值去查表。

假设当前数据包的MAC地址是 `0x112233445566`，那就拿 `0x112233445566` 去查。

------

**第二步：查什么？**

```cpp
const entries = {
    (0x112233445566) : Set_dstAddr();
}
```

表里只有一条固定规则：

- **如果MAC等于 `0x112233445566`**，就执行 `Set_dstAddr()`

------

**第三步：查到没有？**

```text
查表过程：

拿 hdr.eth.dstAddr = 0x112233445566 去查
    ↓
和 const entries 里的 0x112233445566 对比
    ↓
相等！匹配上了！
    ↓
执行 Set_dstAddr()
```

**如果数据包的MAC不是 `0x112233445566`**：

```text
拿 hdr.eth.dstAddr = 0xabcdef123456 去查
    ↓
和 const entries 里的 0x112233445566 对比
    ↓
不相等！没匹配上！
    ↓
执行 default_action = NoAction()
（什么都不做）
```

------

**三、`Set_dstAddr()` 执行时做了什么？**

```cpp
action Set_dstAddr() {
    hdr.eth.dstAddr = 0xaabbccddee02;
    standard_metadata.egress_spec = 0x2;
}
```

**逐行执行：**

| 行    | 做什么                                                       |
| :---- | :----------------------------------------------------------- |
| 第1行 | 把数据包的 `dstAddr` 从 `0x112233445566` 改成 `0xaabbccddee02` |
| 第2行 | 把 `egress_spec` 设为 `2`（意思是：这个包从端口2发出去）     |

------

**四、完整运行流程（按时间顺序）**

```text
【数据包从网口进入交换机】

第1步：Parser 解析
       提取以太网头，填到 hdr.eth 里
       hdr.eth.dstAddr = 0x112233445566  ← 就是这个数据包的目的MAC

第2步：进入 MyIngress control
       执行 apply { mac_match_tbl.apply(); }

第3步：mac_match_tbl.apply() 查表
       拿 hdr.eth.dstAddr = 0x112233445566
       去 const entries 里找有没有匹配的

第4步：找到匹配！
       (0x112233445566) : Set_dstAddr()
       所以执行 Set_dstAddr()

第5步：Set_dstAddr() 执行
       hdr.eth.dstAddr = 0xaabbccddee02  ← 目的MAC被改了
       standard_metadata.egress_spec = 2  ← 标记"从端口2出去"

第6步：MyIngress 结束

第7步：Deparser 组装报文
       从 hdr.eth 读取数据，组装成bit流
       看到 egress_spec = 2，从端口2发出

【数据包从端口2出去了】
```

------

**五、代码中各变量含义**

| 变量                            | 类型                  | 含义                                 |
| :------------------------------ | :-------------------- | :----------------------------------- |
| `hdr`                           | `headers_t`           | 当前数据包的所有头部（以太网、IP等） |
| `hdr.eth`                       | `Ethernet_h`          | 当前数据包的以太网头部               |
| `hdr.eth.dstAddr`               | `bit<48>`             | 以太网头部的目的MAC地址              |
| `standard_metadata`             | `standard_metadata_t` | 交换机标准元数据                     |
| `standard_metadata.egress_spec` | `PortId_t`            | 告诉交换机从哪个端口发出             |

------

**六、一句话总结**

```text
数据包进来 → 查表（用目的MAC查）→ 匹配就改MAC并设端口 → 不匹配就不动 → 发出去
```

------

**七、常见问题**

**Q：代码是按顺序执行的吗？**

```cpp
control MyIngress(...) {
    action Set_dstAddr() { ... }    // ← 只是定义，不执行
    action NoAction() { ... }       // ← 只是定义，不执行
    
    table mac_match_tbl { ... }     // ← 只是定义，不执行
    
    apply {
        mac_match_tbl.apply();      // ← 只有这一行真正执行！
    }
}
```

**真正执行的只有 `apply` 里面的代码！** 上面的 `action` 和 `table` 只是定义，告诉程序"如果匹配上了就执行这个"。

**Q：如果数据包的MAC不是 `0x112233445566` 呢？**

→ 查不到匹配，执行 `default_action = NoAction()`，什么都不做，原样发出去。

**Q：`hdr.eth.dstAddr` 的值从哪里来？**

→ Parser 从原始报文中提取出来的。原始bit流的前6字节是目的MAC，Parser 把它填到 `hdr.eth.dstAddr` 里。

------

**核心：每个数据包来，查一次表，匹配就执行action，不匹配就执行默认动作。**



### 3.7.1 control 的定义

control 是没有返回值的，格式如下：

```cpp
control MyIngress(inout headers hdr, inout metadata meta, inout standard_metadata_t standard_metadata)
{
    // control 主体
    action set_dst_addr(bit<9> port) {
        hdr.eth.dstAddr = 0xAAABBCCDDEE02;
        standard_metadata.egress_spec = 0x2;
	}
    
}
```

- 参数必须指定方向（`in`、`out`、`inout`）
- `hdr`：正在处理的报文头部（用户定义）
- `meta`：元数据（用户定义）
- `standard_metadata`：标准元数据（由 Architecture/Target 定义）

**v1model 架构中 standard_metadata_t 的定义**

```cpp
// v1model.p4
struct standard_metadata_t {
    PortId_t ingress_port;          // 入端口
    PortId_t egress_spec;            // 出端口（在 ingress 流水线中设置）
    PortId_t egress_port;
    bit<32> instance_type;
    bit<32> packet_length;
    bit<32> enq_timestamp;
    bit<19> enq_qdepth;
    bit<32> deq_timedelta;
    bit<19> deq_qdepth;
    bit<48> ingress_global_timestamp;
    bit<48> egress_global_timestamp;
    bit<16> mcast_grp;
    bit<16> egress_rid;
    bit<1> checksum_error;
    error parser_error;
    bit<3> priority;
}
```

------

### 3.7.2 action

**基本概念**

`action` 是匹配-动作表中匹配命中后执行的代码片段，可以修改**报文头部**和**元数据**。

```cpp
action action_name(参数) {
    语句;
}
```

**action 的特点**

| 特点           | 说明                                                      |
| :------------- | :-------------------------------------------------------- |
| 无返回值       | 类似于没有返回值的函数                                    |
| 参数需指定方向 | 无方向参数必须在参数列表末尾（由控制面提供）              |
| 作用域         | 全局声明则全局可用；control 内声明则仅在该 control 内可用 |

**action 示例**

```cpp
action set_dst_addr(bit<9> port) {
    hdr.eth.dstAddr = 0xAAABBCCDDEE02;
    standard_metadata.egress_spec = 0x2;
}
```

**action 的调用方式**

| 方式     | 说明                              |
| :------- | :-------------------------------- |
| 隐式调用 | 由 table 在匹配-动作处理期间执行  |
| 显式调用 | 由 control 或其他 action 直接调用 |

**action 的使用限制**

1. 可以使用 control 的参数和 action 自身的参数
2. 无方向参数必须出现在参数列表末尾（由控制面或 `default_action`/`const entries` 提供）
3. 参数不能有 `extern` 类型
4. **不能使用 `switch` 或 `select` 语句**（某些 Target 支持 `if`）
5. 可以使用 `return` 立即终止

**action 与函数的区别**

| 特性     | action              | 函数     |
| :------- | :------------------ | :------- |
| 返回值   | ❌ 不能              | ✅ 可以有 |
| 条件语句 | 某些 Target 不支持  | ✅ 支持   |
| 调用方式 | 隐式（table）+ 显式 | 仅显式   |

------

### 3.7.3 table

**基本概念**

`table` 定义匹配条件、匹配方法及匹配命中后执行的动作。

```cpp
table table_name {
    key = { 字段1 : 匹配方法1; 字段n : 匹配方法n; }
    actions = { action_1; action_n; }
    default_action = NoAction();    // 可选
    size = 1024;                    // 可选
    const entries = { ... };        // 可选，常量表项
    implementation = ActionProfile(1024); // 可选
}
```

**1. key（匹配键）**

key 用 `e : m` 形式描述：

- `e`：要匹配的字段表达式
- `m`：匹配算法（`match_kind`）

**P4 核心库中的 match_kind**

```cpp
match_kind {
    exact,      // 精确匹配
    ternary,    // 三态/掩码匹配
    lpm         // 最长前缀匹配
}
```

| 匹配方式  | 说明                                             |
| :-------- | :----------------------------------------------- |
| `exact`   | 输入值必须完全等于匹配项                         |
| `ternary` | 控制面写入 value 和 mask，`输入 & mask == value` |
| `lpm`     | 选择具有最长前缀的匹配项                         |

**示例：**

```cpp
table fwd_tbl {
    key = {
        ipv4header.dstAddress : ternary;
        ipv4header.version : exact;
    }
}
```

> **注意：所有 key 字段都匹配成功，该表项才算匹配成功！**

**特殊情况：** table 可以没有 `key`，此时只包含一个默认动作，对所有报文执行该动作。

------

**2. actions（动作列表）**

- `actions` 包含该 table 中所有动作的列表（包括 `default_action`）
- 同一 table 中 action 必须有不同的名字
- **有方向(in out inout)参数必须在 action 列表中绑定**（使用变量或常量）
- **无方向参数由控制面提供**，不需要绑定
- 在 P4 语言中，**当一个参数没有写任何方向（`in`/`out`/`inout`）时，它默认就是 `in` 方向**。

**参数绑定示例：**

```cpp
action a(in bit<32> x) { /* 代码省略 */ }
action b(inout bit<32> x, bit<8> data) { /* 代码省略 */ }

bit<32> z;

table t {
    actions = {
        a;         // ❌ 错误：x 必须被绑定,in bit<32> x有方向需要绑定
        a(5);      // ✅ 正确：x 绑定为 5,int bit<32> x
        b(z);      // ✅ 正确：x 绑定为 z
        b(z, 3);   // ❌ 错误：无方向参数 data 不能用 3 绑定
        		   // 因为 data 是第二个参数，P4 语法规定只有第一个action参数可以用常量绑定
        b();       // ❌ 错误：x 必须被绑定
    }
}
```

------

**3. default_action（默认动作）**

- 当 table 查找**未命中**时执行
- **可选**
- 必须出现在 `actions` 列表的**最后边**
- 无方向参数可由控制面改变
- 如果未指定且查找未命中 → **对报文不做任何修改**

------

**4. 常量表项（const entries）**

- 不可变，只能读取，不能由控制面更改或删除
- 可用于实现固定算法，节省计算资源

```cpp
const entries = { (0x112233445566) : set_dst_addr(); }
```

------

**5. size（表大小）**

- 提示编译器表项的数量（必须是常量表达式还有整数）
- 实际能插入的表项数可能小于 `size`（受哈希冲突等限制）

------

**6. implementation（实现属性）**

- 用于指定 extern 组件（如 `ActionProfile`）
- 可通过引入间接层共享 action，减少存储容量

```cpp
extern ActionProfile {
    ActionProfile(bit<32> size);
}

table tbl {
    actions = { /* ... */ }
    implementation = ActionProfile(1024);
}
```

------

**7. table 的调用**

调用 `table.apply()` 方法执行匹配-动作。

```cpp
apply {
    mac_match_tbl.apply();
}
```

**8. apply 方法的返回值**

编译器为每个 `table T` 自动合成 `apply_result(T)` 结构：

```cpp
enum action_list(T) {
    // table T 中每一个 action 的枚举
}

struct apply_result(T) {
    bool hit;           // 是否命中
    bool miss;          // 是否未命中
    action_list(T) action_run;  // 执行了哪个 action
}
```

**使用示例：**

```cpp
// 判断命中/未命中
if (mac_match_tbl.apply().hit) {
    // 命中
} else {
    // 未命中
}

// switch 判断执行了哪个 action
switch (mac_match_tbl.apply().action_run) {
    set_dst_addr: { return; }
    default: { /* 代码省略 */ }
}
```



------

### 3.7.4 control 的调用

- control 可以由 **package** 调用，也可以由**另一个 control** 调用

**示例：**

```cpp
control Callee(inout IPv4 ipv4) {
    /* 代码省略 */
}

control Caller(inout Headers h) {
    Callee() instance;
    apply {
        instance.apply(h.ipv4);
    }
}
```

**control 使用限制**

1. **不能实例化 parser**
2. 除 `exit` 语句外不支持其他异常控制流
3. 不支持 `verify`、`reject` 等语句

------

## 3.8 `parser` 的作用

| 功能           | 说明                                     |
| :------------- | :--------------------------------------- |
| 提取报文头部   | 将原始bit流填入 header 结构              |
| 设置 metadata  | 根据报文信息填充元数据，供后续流水线使用 |
| 校验报文合法性 | 检查 version、checksum 等字段是否正确    |
| 丢弃非法报文   | 校验失败时 reject 报文                   |

> **parser 是有限状态机（FSM），底层由 TCAM 实现。**



### 完整 `parser` 示例

```cpp

    Checksum16() ck;

    state start {
        b.extract(p.ethernet);
        transition select(p.ethernet.etherType) {
            0x0800: parse_ipv4;
            0x0806: parse_arp;
            default: accept;

        verify(p.ipv4.ihl == 4w5, error.IPv4OptionsNotSupported);
        ck.clear();

        verify(ck.get() == 16w0, error.IPv4ChecksumError);
        transition select(p.ipv4.protocol) {
            8w6  : parse_tcp;
            8w17 : parse_udp;
            _    : accept;
        }
    }

    state parse_tcp {
        b.extract(p.tcp);
        transition accept;
    }

    state parse_udp {
        b.extract(p.udp);
        transition accept;
    }

    state parse_arp {
        b.extract(p.arp);
        transition accept;
    }
}
```

### 3.8.1 parser 的定义

```cpp
parser TopParser(packet_in b, out Parsed_packet p) {
    // 局部变量（extern 组件等）
    Checksum16() ck;

    state start {
        b.extract(p.ethernet);
        transition select(p.ethernet.etherType) {
            0x0800: parse_ipv4;
            // 没有 default → 非 IPv4 报文被丢弃
        }
    }

    state parse_ipv4 {
        b.extract(p.ip);
        verify(p.ip.version == 4w4, error.IPv4IncorrectVersion);
        verify(p.ip.ihl == 4w5, error.IPv4OptionsNotSupported);
        ck.clear();
        ck.update(p.ip);
        verify(ck.get() == 16w0, error.IPv4ChecksumError);
        transition accept;
    }
}
```



### 3.8.2 parser 的组成部分

| 组成部分      | 说明                                   |
| :------------ | :------------------------------------- |
| 名字          | 如 `TopParser`                         |
| 参数列表      | 至少一个 `packet_in` 类型的参数        |
| 局部变量      | 在 parser 内部定义的变量或 extern 组件 |
| 状态（state） | 用 `state` 关键字定义的状态            |
| `start` 状态  | **必须存在且唯一**，parser 从这里开始  |
| `accept` 状态 | 系统预定义，解析成功                   |

### 3.8.3 parser 中的语句

**1）select 语句（条件跳转）**

> **parser 中不能使用 `if` 语句，用 `select` 代替！**

```cpp
state parse_ipv4 {
    select (p.ipv4.protocol) {
        8w6  : parse_tcp;      // 单个值
        8w17 : parse_udp;      // 单个值
        _    : accept;         // 通配（任何值）
    }
}
```

**select 标签支持的格式：**

| 格式   | 示例                 | 说明                         |
| :----- | :------------------- | :--------------------------- |
| 单个值 | `8w6`                | 精确匹配                     |
| 忽略值 | `_`                  | 通配所有值（相当于 default） |
| 掩码值 | `16w0 &&& 16w0xFc00` | `value & mask == a`          |
| 范围值 | `4w6 .. 4w15`        | 包含 6 到 15 的所有整数      |
| 元组   | `(4w0x5, 8w0x1)`     | 多个字段组合匹配             |

**select 标签值可以 "重叠"（重复或包含关系）！** 这是与 C 语言 switch 的重要区别。

**匹配顺序：** 从上到下，由第一个匹配的标签提供结果状态。如果没有匹配，触发运行时错误 `error.NoMatch`。

**2）transition 语句（状态跳转）**

`transition` 实现状态机转移，类似 C 语言的 `goto`。

```cpp
state start {
    transition parse_ethernet;    // 跳转到 parse_ethernet
}

state parse_ethernet {
    pkt.extract(hdr.ethernet);
    transition parse_ipv4;        // 跳转到 parse_ipv4
}

state parse_ipv4 {
    pkt.extract(hdr.ipv4);
    transition accept;            // 跳转到 accept（系统状态）
}
```

> **注意：`select` 语句中省略了 `transition` 关键字！**

**3）verify 语句（条件校验）**

类似 C 语言的 `assert`。

```cpp
verify(p.ip.version == 4w4, error.IPv4IncorrectVersion);
```

| 条件             | 结果                                              |
| :--------------- | :------------------------------------------------ |
| 表达式为 `true`  | 继续执行后续语句                                  |
| 表达式为 `false` | 返回错误码，设置 parser 为 `reject`，立即终止解析 |

> **`verify` 语句只能在 parser 中使用！**

### 3.8.4 parser value set（动态匹配集）

**用途：** 让 parser 的匹配规则可以在运行时由控制面动态修改。

```cpp
struct vsk_t {
    @match(exact)
    bit<16> port;
}

value_set<vsk_t>(4) pvs;    // 定义包含4个元素的集合

    pvs: runtime_defined_port;    // 命中 pvs 集合中的值

}
```

**报文组装过程**

```text
原始报文解析时：
[以太网头][IPv4头][TCP头][数据]

Deparser 组装时（按 emit 顺序）：
第1步：放入以太网头    → [以太网头]
第2步：放入IPv4头     → [以太网头][IPv4头]
第3步：放入TCP头      → [以太网头][IPv4头][TCP头]
第4步：自动追加数据    → [以太网头][IPv4头][TCP头][数据]
                      ↑
                 完整报文！
```



**最终发出报文的顺序**

```text
发出顺序：以太网头 → IPv4头 → TCP头 → 数据
```

**`emit()` 的顺序就是报文中字段出现的顺序！**



**emit 对 header 有效性的检查**

```cpp
// 仅当 header valid = true 时，才会被添加到报文
b.emit(hdr.ethernet);   // 如果 ethernet 是 valid → 添加；否则跳过
b.emit(hdr.ipv4);       // 如果 ipv4 是 valid → 添加；否则跳过
b.emit(hdr.tcp);        // 如果 tcp 是 valid → 添加；否则跳过
```



**不同协议报文的 emit 行为**

**场景1：IPv4 TCP 报文**

```text
hdr.ethernet.valid = true  →  emit 以太网头 ✅
hdr.ipv4.valid = true      →  emit IPv4头 ✅
hdr.tcp.valid = true       →  emit TCP头 ✅
hdr.udp.valid = false      →  不 emit
hdr.arp.valid = false      →  不 emit

最终发出：[以太网头][IPv4头][TCP头][数据]
```

**场景2：IPv4 UDP 报文**

```text
hdr.ethernet.valid = true  →  emit 以太网头 ✅
hdr.ipv4.valid = true      →  emit IPv4头 ✅
hdr.tcp.valid = false      →  不 emit
hdr.udp.valid = true       →  ⚠️ 代码里没写 emit(hdr.udp)！
hdr.arp.valid = false      →  不 emit

最终发出：[以太网头][IPv4头][数据]  ← UDP头丢失了！
```

**代码中只 emit 了 TCP，如果遇到 UDP 报文，UDP 头会丢失！**



**支持多协议的 Deparser**

```cpp
// 支持 UDP 和 ARP 的 Deparser
control MyDeparser(packet_out b, in headers hdr) {
    apply {
        b.emit(hdr.ethernet);
        b.emit(hdr.ipv4);
        b.emit(hdr.tcp);     // TCP 头
        b.emit(hdr.udp);     // UDP 头
        b.emit(hdr.arp);     // ARP 头
        // 因为每次报文只有一个协议，所以只有一个有效
        // 无效的 header 会被自动跳过
    }
}
```

**执行流程**

```text
TCP 报文：有效的是 tcp，emit tcp 时添加；udp 无效，跳过；arp 无效，跳过
UDP 报文：有效的是 udp，emit udp 时添加；tcp 无效，跳过；arp 无效，跳过
ARP 报文：有效的是 arp，emit arp 时添加；tcp 无效，跳过；udp 无效，跳过
```

**利用 `emit()` 自动检查 valid 位的特性，多个 `emit()` 只添加有效的那个！**

### 3.10.2 v1model 架构的 V1Switch

```cpp
package V1Switch<H, M>(
    Parser<H, M> p,
    VerifyChecksum<H, M> vr,
    Ingress<H, M> ig,
    Egress<H, M> eg,
    ComputeChecksum<H, M> ck,
    Deparser<H> dep
);
```

| 部分       | 含义                                         |
| :--------- | :------------------------------------------- |
| `package`  | 关键字，声明这是一个包                       |
| `V1Switch` | package 的名字                               |
| `<H, M>`   | 泛型参数（表示 header 类型和 metadata 类型） |
| `(...)`    | 6 个组件参数，按顺序传入                     |

**v1model 架构定义了 6 个可编程组件，`package` 要求必须全部提供。**

**v1model 完整流水线**

```
┌──────────┐ ┌──────────────┐ ┌──────────┐ ┌─────────┐ ┌───────────────┐ ┌──────────┐
│  parser  │→│ verify       │→│ ingress  │→│ egress  │→│ compute       │→│ deparser │
│          │ │ checksum     │ │          │ │         │ │ checksum      │ │          │
└──────────┘ └──────────────┘ └──────────┘ └─────────┘ └───────────────┘ └──────────┘
     ↑              ↑              ↑           ↑              ↑              ↑
  第1个参数      第2个参数      第3个参数    第4个参数      第5个参数      第6个参数
```



**数据包按顺序流经这 6 个组件！**

**如果不用 package 会怎样？**

```cpp
// ❌ 没有 package，这些组件只是各自独立存在
parser MyParser(...) { ... }
control MyIngress(...) { ... }
control MyDeparser(...) { ... }

// 编译器不知道：
// 1. 从哪个组件开始？
// 2. 组件之间的顺序是什么？
// 3. 数据怎么在组件之间传递？

// ✅ 有了 package，一切明确
V1Switch(MyParser(), MyIngress(), MyDeparser()) main;
```



**没有 package，编译器不知道程序的"入口"在哪里，也不知道各个组件如何连接。**

**数据怎么在组件之间传递？**

由 **package 内部自动完成**！

```text
数据包进入
    ↓
Parser 解析 → 输出 hdr 和 meta
    ↓
（hdr, meta 自动传入）VerifyChecksum
    ↓
（hdr, meta 自动传入）Ingress
    ↓
（hdr, meta 自动传入）Egress
    ↓
（hdr, meta 自动传入）ComputeChecksum
    ↓
（hdr 自动传入）Deparser
    ↓
数据包发出
```



**程序员只需要定义每个组件的逻辑，数据传递由 package 自动管理。**

**package 实例化规则**

```cpp
// ✅ 正确：所有参数都传入
V1Switch(
    MyParser(),
    MyVerifyChecksum(),
    MyIngress(),
    MyEgress(),
    MyComputeChecksum(),
    MyDeparser()
) main;

// ❌ 错误：参数数量不对（少于6个）
V1Switch(
    MyParser(),
    MyIngress(),
    MyDeparser()
) main;

// ❌ 错误：参数类型不对（传错类型）
V1Switch(
    MyParser(),      // 正确
    MyIngress(),     // 期望 VerifyChecksum，但传了 Ingress
    ...
) main;
```



**package 与 main 的关系**

```cpp
V1Switch(...) main;
```

| 部分            | 含义                   |
| :-------------- | :--------------------- |
| `V1Switch(...)` | 创建一个 V1Switch 对象 |
| `main`          | 程序入口点（固定名称） |
| `;`             | 结束声明               |

**`main` 是固定关键字，表示这是程序的顶层入口。每个 P4 程序必须有且只有一个 `main`！**



## 各组件语句使用限制总结

| 语句         | parser   | control  | action           |
| :----------- | :------- | :------- | :--------------- |
| `if` 条件    | ❌ 不允许 | ✅ 允许   | 部分 Target 允许 |
| `select`     | ✅ 允许   | ❌ 不允许 | ❌ 不允许         |
| `switch`     | ❌ 不允许 | ✅ 允许   | ❌ 不允许         |
| `transition` | ✅ 允许   | ❌ 不允许 | ❌ 不允许         |
| `verify`     | ✅ 允许   | ❌ 不允许 | ❌ 不允许         |
| `return`     | ❌ 不允许 | ✅ 允许   | ✅ 允许           |
| `exit`       | ❌ 不允许 | ✅ 允许   | ❌ 不允许         |

```cpp
// 1. include 核心库
```
