```bash
pkill -9 -f simple_switch 2>/dev/null

timeout 20 mn -c 2>/dev/null
for intf in $(ip link show 2>/dev/null | grep -oE "s[0-9]-eth[0-9]+|h[0-9]+-eth[0-9]+" | sort -u); do
    ip link del "$intf" 2>/dev/null
done
echo "mininet cleaned"
```



```bash
# 清理并启动
pkill -9 Xvfb x11vnc openbox 2>/dev/null; rm -f /tmp/.X0-lock /tmp/.X11-unix/X0; mkdir -p /tmp/.X11-unix; chmod 1777 /tmp/.X11-unix; Xvfb :0 -screen 0 1280x800x24 > /tmp/xvfb.log 2>&1 & sleep 2; export DISPLAY=:0; x11vnc -forever -shared -display :0 -rfbport 5900 -nopw > /tmp/x11vnc.log 2>&1 & sleep 1; openbox > /tmp/openbox.log 2>&1 &
```



使用流程
1. 连接 VNC（WSL 终端执行，窗口会显示到 Windows 桌面）：

  ```bash
  vncviewer 172.18.0.2:5900
  ```

2. 让 xterm 窗口出现（连上后先看到空桌面是正常的）：
# 进容器
```bash
docker exec -it p4-bmv2 bash
cd /workspace/tutorials/exercises/basic
make run      # 如果还没启动 mininet
```



# 在 mininet 提示符里打开 4 个主机终端
```bash
mininet> xterm h1 h11 h2 h22
这 4 个 xterm 窗口会渲染到容器的 Xvfb（:0）上，x11vnc 把它转发到 5900 端口——VNC 里立刻就能看到它们，分别在里面跑 receive.py / send.py。
常用命令速查
操作	命令
连接 VNC	vncviewer 172.18.0.2:5900
断开	关窗口即可
查看 VNC 是否在跑	docker exec p4-bmv2 pgrep -a x11vnc
容器 IP 变了再查	docker exec p4-bmv2 hostname -I
提示：如果 xterm h1... 报找不到主机的错，先确认 mininet 拓扑里主机名是 h1 h11 h2 h22（不同练习拓扑不同，看 topology.json 确认）。
```
